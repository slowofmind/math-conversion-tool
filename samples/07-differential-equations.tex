\section*{Second-Order Linear ODEs}

\begin{theorem*}[Characteristic Equation Method]
For the constant-coefficient equation $ay'' + by' + cy = 0$, the characteristic equation is $ar^2 + br + c = 0$.
\begin{itemize}
  \item If $r_1 \neq r_2$ are real: $y = c_1 e^{r_1 t} + c_2 e^{r_2 t}$.
  \item If $r_1 = r_2 = r$: $y = (c_1 + c_2 t) e^{rt}$.
  \item If $r = \alpha \pm \beta i$: $y = e^{\alpha t}(c_1 \cos \beta t + c_2 \sin \beta t)$.
\end{itemize}
\end{theorem*}

\begin{enumerate}[resume]
\item Find the general solution to each ODE.
\begin{enumerate}[(a)]
  \item $y'' - 5y' + 6y = 0$
  \item $y'' + 4y' + 4y = 0$
  \item $y'' + 9y = 0$
  \item $y'' - 2y' + 5y = 0$
\end{enumerate}

\item Solve the initial value problem:
\[
  y'' + 2y' + 5y = 0, \qquad y(0) = 1, \quad y'(0) = -3.
\]

\item Use the method of undetermined coefficients to find a particular solution, then write the general solution.
\begin{enumerate}[(a)]
  \item $y'' - 3y' + 2y = 4e^{3t}$
  \item $y'' + y = \sin t$ \quad \emph{(Watch for resonance!)}
\end{enumerate}
\end{enumerate}

\section*{Laplace Transforms}

\begin{definition*}
The \textbf{Laplace transform} of $f(t)$ is
\[
  \Lap\{f(t)\} = F(s) = \int_0^{\infty} e^{-st} f(t) \, dt.
\]
\end{definition*}

\textbf{Key transform pairs:}
\begin{alignat*}{3}
  &\Lap\{1\} = \frac{1}{s}, \qquad
  &&\Lap\{t^n\} = \frac{n!}{s^{n+1}}, \qquad
  &&\Lap\{e^{at}\} = \frac{1}{s - a}, \\[6pt]
  &\Lap\{\sin bt\} = \frac{b}{s^2 + b^2}, \qquad
  &&\Lap\{\cos bt\} = \frac{s}{s^2 + b^2}, \qquad
  &&\Lap\{f'(t)\} = s F(s) - f(0).
\end{alignat*}

\begin{enumerate}[resume]
\item Use Laplace transforms to solve the initial value problem:
\[
  y'' + 4y = \sin 2t, \qquad y(0) = 0, \quad y'(0) = 1.
\]

\item Find the inverse Laplace transform:
\begin{enumerate}[(a)]
  \item $\displaystyle \invLap\left\{ \frac{3s + 5}{s^2 + 4s + 13} \right\}$

  \item $\displaystyle \invLap\left\{ \frac{s}{(s-1)(s^2 + 1)} \right\}$
\end{enumerate}
\end{enumerate}

\section*{Systems of ODEs}

\begin{enumerate}[resume]
\item Find the general solution to the system $\mathbf{x}' = A\mathbf{x}$ where
\[
  A = \begin{pmatrix} 1 & 3 \\ 3 & 1 \end{pmatrix}.
\]

\item Solve the system
\[
  \frac{d}{dt} \begin{pmatrix} x \\ y \end{pmatrix}
  = \begin{pmatrix} 0 & 1 \\ -4 & 0 \end{pmatrix}
    \begin{pmatrix} x \\ y \end{pmatrix}, \qquad
  \begin{pmatrix} x(0) \\ y(0) \end{pmatrix} = \begin{pmatrix} 1 \\ 2 \end{pmatrix}.
\]

\item Classify the origin as a critical point (node, saddle, spiral, center) for each matrix and determine stability.
\begin{enumerate}[(a)]
  \item $A = \begin{pmatrix} -1 & 2 \\ 0 & -3 \end{pmatrix}$ \quad ($\Tr(A) = -4$, $\det(A) = 3$)

  \item $A = \begin{pmatrix} 1 & -5 \\ 1 & -1 \end{pmatrix}$
\end{enumerate}
\end{enumerate}
