\section*{Review of Key Formulas}

Recall the following integration formulas:
\begin{gather}
  \int x^n \, dx = \frac{x^{n+1}}{n+1} + C \quad (n \neq -1), \qquad
  \int e^x \, dx = e^x + C, \qquad
  \int \frac{1}{x} \, dx = \ln|x| + C. \\[6pt]
  \int \sin x \, dx = -\cos x + C, \qquad
  \int \cos x \, dx = \sin x + C, \qquad
  \int \sec^2 x \, dx = \tan x + C.
\end{gather}

\textbf{Integration by Parts:}
\[
  \int u \, dv = uv - \int v \, du.
\]

\section*{Part I: $u$-Substitution}

\begin{enumerate}
\item Evaluate each integral using substitution.
\begin{enumerate}[(a)]
  \item $\displaystyle \int 3x^2 \cos(x^3) \, dx$

  \item $\displaystyle \int \frac{e^{\sqrt{x}}}{\sqrt{x}} \, dx$

  \item $\displaystyle \int_0^1 \frac{x}{(1 + x^2)^2} \, dx$

  \item $\displaystyle \int \frac{\ln x}{x} \, dx$
\end{enumerate}
\end{enumerate}

\section*{Part II: Integration by Parts}

\begin{enumerate}[resume]
\item Evaluate each integral using integration by parts.
\begin{enumerate}[(a)]
  \item $\displaystyle \int x e^{2x} \, dx$

  \item $\displaystyle \int x^2 \sin x \, dx$

  \item $\displaystyle \int \ln x \, dx$

  \item $\displaystyle \int e^x \cos x \, dx$
\end{enumerate}

\item \textbf{Worked Example.} Here is a detailed solution for $\displaystyle \int x^2 e^x \, dx$, applying integration by parts twice:
\begin{align*}
  \int x^2 e^x \, dx
    &= \underbrace{x^2}_{u} \cdot \underbrace{e^x}_{v}
       \;-\; \int \underbrace{e^x}_{v} \cdot \underbrace{2x \, dx}_{du} \\[6pt]
    &= x^2 e^x - 2\!\int x e^x \, dx \\[6pt]
    &= x^2 e^x - 2\!\left[\, x e^x - \int e^x \, dx \,\right] \\[6pt]
    &= x^2 e^x - 2x e^x + 2e^x + C \\[6pt]
    &= e^x\bigl(x^2 - 2x + 2\bigr) + C.
\end{align*}
\end{enumerate}

\section*{Part III: Partial Fractions}

\begin{enumerate}[resume]
\item Evaluate each integral using partial fraction decomposition.
\begin{enumerate}[(a)]
  \item $\displaystyle \int \frac{5x + 3}{(x+1)(x-2)} \, dx$

  \item $\displaystyle \int \frac{3x^2 + 2}{x(x^2 + 1)} \, dx$

  \item $\displaystyle \int \frac{x^3 + 2}{x^2 - 1} \, dx$ \quad \emph{(Hint: Perform polynomial long division first.)}
\end{enumerate}

\item Show the partial fraction decomposition (without evaluating the integral):
\[
  \frac{2x^2 + x + 4}{(x+1)(x^2 + 4)} = \frac{A}{x+1} + \frac{Bx + C}{x^2 + 4}.
\]
Find $A$, $B$, and $C$.
\end{enumerate}
