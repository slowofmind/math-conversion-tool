\begin{definition*}
We write $\displaystyle \lim_{x \to a} f(x) = L$ if for every $\varepsilon > 0$, there exists $\delta > 0$ such that
\[
  0 < |x - a| < \delta \implies |f(x) - L| < \varepsilon.
\]
\end{definition*}

\section*{Part I: Computing Limits}

\begin{enumerate}
\item Evaluate each limit, or explain why it does not exist.
\begin{enumerate}[(a)]
  \item $\displaystyle \lim_{x \to 3} \frac{x^2 - 9}{x - 3}$

  \item $\displaystyle \lim_{x \to 0} \frac{\sin(5x)}{3x}$

  \item $\displaystyle \lim_{x \to 0^+} \frac{1}{x} - \frac{1}{|x|}$

  \item $\displaystyle \lim_{x \to 4} \frac{\sqrt{x} - 2}{x - 4}$

  \item $\displaystyle \lim_{x \to \infty} \frac{3x^2 - x + 4}{5x^2 + 2x - 1}$

\item Evaluate the limit using L'H\^{o}pital's Rule where appropriate.
\begin{enumerate}[(a)]
  \item $\displaystyle \lim_{x \to 0} \frac{e^x - 1 - x}{x^2}$

  \item $\displaystyle \lim_{x \to 0^+} x \ln x$

  \item $\displaystyle \lim_{x \to \infty} \left(1 + \frac{3}{x}\right)^x$
\end{enumerate}
\end{enumerate}

\section*{Part II: One-Sided Limits and Continuity}

\begin{enumerate}[resume]
\item Let
\[
  f(x) = \begin{cases}
    x^2 + 1 & \text{if } x < 1, \\[4pt]
    3       & \text{if } x = 1, \\[4pt]
    4 - x   & \text{if } x > 1.
  \end{cases}
\]
\begin{enumerate}[(a)]
  \item Find $\displaystyle \lim_{x \to 1^-} f(x)$ and $\displaystyle \lim_{x \to 1^+} f(x)$.
  \item Does $\displaystyle \lim_{x \to 1} f(x)$ exist? Explain.
  \item Is $f$ continuous at $x = 1$? If not, classify the discontinuity as removable, jump, or infinite.
\end{enumerate}

\item Find the value of $c$ that makes the function continuous everywhere:
\[
  g(x) = \begin{cases}
    \dfrac{x^2 - 4}{x - 2} & \text{if } x \neq 2, \\[8pt]
    c & \text{if } x = 2.
  \end{cases}
\]

\item For each function, find all points of discontinuity and classify each as removable, jump, or infinite.
\begin{enumerate}[(a)]
  \item $\displaystyle f(x) = \frac{x^2 + x - 6}{x^2 - 4}$

  \item $\displaystyle g(x) = \frac{|x - 1|}{x - 1}$
\end{enumerate}
\end{enumerate}

\section*{Part III: The Intermediate Value Theorem}

\begin{theorem}[Intermediate Value Theorem]
If $f$ is continuous on $[a, b]$ and $N$ is any number between $f(a)$ and $f(b)$, then there exists at least one $c \in (a, b)$ such that $f(c) = N$.
\end{theorem}

\begin{enumerate}[resume]
\item Use the Intermediate Value Theorem to show that $x^3 + x - 1 = 0$ has a solution in the interval $(0, 1)$.

\item Suppose $f$ is continuous on $[0, 4]$ with $f(0) = -2$ and $f(4) = 5$. Prove there exists $c \in (0, 4)$ such that $f(c) = \pi$.

\item True or False. Justify your answer with a proof or counterexample.
\begin{enumerate}[(a)]
  \item If $\displaystyle \lim_{x \to a} f(x)$ exists and $\displaystyle \lim_{x \to a} g(x)$ does not exist, then $\displaystyle \lim_{x \to a} [f(x) + g(x)]$ does not exist.
  \item If neither $\displaystyle \lim_{x \to a} f(x)$ nor $\displaystyle \lim_{x \to a} g(x)$ exists, then $\displaystyle \lim_{x \to a} [f(x) + g(x)]$ does not exist.
\end{enumerate}
\end{enumerate}
