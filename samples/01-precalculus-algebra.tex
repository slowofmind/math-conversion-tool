\section*{Polynomials and Rational Expressions}

\begin{enumerate}
\item Factor each polynomial completely.
\begin{enumerate}[(a)]
  \item $6x^3 - 54x$
  \item $x^4 - 16$
  \item $2x^3 + 5x^2 - 3x$
  \item $x^3 + 27$
\end{enumerate}

\item Simplify each rational expression. State any restrictions on the variable.
\begin{enumerate}[(a)]
  \item $\displaystyle \frac{x^2 - 9}{x^2 + 5x + 6}$

  \item $\displaystyle \frac{2x^2 + 7x + 3}{x^2 - 9} \cdot \frac{x - 3}{2x + 1}$

  \item $\displaystyle \frac{1}{x + 2} + \frac{3}{x - 1}$
\end{enumerate}

\item Solve for $x$:
\begin{enumerate}[(a)]
  \item $|2x - 5| = 7$
  \item $|3x + 1| \leq 10$
  \item $x^2 - 4x - 12 > 0$
\end{enumerate}
\end{enumerate}

\section*{Functions and Their Properties}

\begin{enumerate}[resume]
\item Let $f(x) = x^2 - 4$ and $g(x) = \sqrt{x + 3}$.
\begin{enumerate}[(a)]
  \item Find the domain of $f$, $g$, and $f \circ g$.
  \item Compute $(f \circ g)(x)$ and simplify.
  \item Compute $(g \circ f)(x)$ and state its domain.
  \item Is $f$ one-to-one? Explain why or why not.
\end{enumerate}

\item For the piecewise function
\[
  h(x) = \begin{cases}
    2x + 1 & \text{if } x < 0, \\
    x^2     & \text{if } 0 \leq x \leq 3, \\
    10 - x  & \text{if } x > 3,
  \end{cases}
\]
evaluate each of the following:
\begin{enumerate}[(a)]
  \item $h(-2)$
  \item $h(0)$
  \item $h(3)$
  \item $h(5)$
\end{enumerate}

\item A ball is thrown upward from the top of a 48-foot building with an initial velocity of 32 feet per second. Its height after $t$ seconds is given by
\[
  s(t) = -16t^2 + 32t + 48.
\]
\begin{enumerate}[(a)]
  \item At what time does the ball reach its maximum height?
  \item What is the maximum height?
  \item When does the ball hit the ground?
\end{enumerate}
\end{enumerate}

\section*{Exponents and Logarithms}

\begin{enumerate}[resume]
\item Simplify each expression. Write your answer with positive exponents only.
\begin{enumerate}[(a)]
  \item $\displaystyle \frac{(2x^3 y^{-2})^2}{4x^{-1} y^5}$

  \item $\displaystyle \left( \frac{27a^6}{8b^{-3}} \right)^{2/3}$
\end{enumerate}

\item Solve for $x$:
\begin{enumerate}[(a)]
  \item $3^{2x-1} = 81$
  \item $\log_2(x+3) + \log_2(x-1) = 5$
  \item $e^{2x} - 5e^x + 6 = 0$
\end{enumerate}
\end{enumerate}
