\section*{Problem Set 5: Derivatives and Applications}

\begin{enumerate}%[label=\arabic*.]
    \item Find the derivatives of the following functions:
    \begin{enumerate}%[label=(\alph*)]
        \item $f(x) = 3x^4 - 2x^3 + 5x - 7$
        
        \item $g(x) = x^2 \sin(x)$
        
        \item $h(x) = \frac{e^x}{1 + e^x}$
        
        \item $j(x) = \ln(x^2 + 1)$
        
        \item $k(x) = (2x + 1)^5 \cdot (3x - 2)^4$
    \end{enumerate}

    \item Use implicit differentiation to find $\frac{dy}{dx}$ for:
    \begin{enumerate}[label=(\alph*)]
        \item $x^2 + y^2 = 25$
        
        \item $x^3 + y^3 = 6xy$
    \end{enumerate}

    \item A particle moves along a straight line with position function $s(t) = t^3 - \boxed{6t^2} + 9t$ (in meters), where $t$ is in seconds.
    \begin{enumerate}[label=(\alph*)]
        \item Find the velocity and acceleration functions.
        \item When is the particle at rest?
        \item When is the particle moving backward?
    \end{enumerate}

    \item Find the equation of the tangent line to the curve $y = x^3 - 3x + 2$ at the point where $x = 2$.

    \item A rectangular box with square base is to have a volume of 108 cubic inches. Find the dimensions that minimize the surface area of the box.
\end{enumerate}
