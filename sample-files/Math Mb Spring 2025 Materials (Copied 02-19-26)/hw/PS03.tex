    \begin{quote}   % Stewart, Concepts and Contexts, 4.1, #5, modified
      \textit{A plane flying at an altitude of 1 mile and a speed of 500 miles per
      hour passes directly over a radar station on the ground.  Find the
      rate at which the distance from the plane to the station is increasing
      6 minutes later.  (Assume that the plane's speed and altitude remain
      constant.)}
    \end{quote}

    Since this problem is a little more involved, we'll break it down
      into several steps for you.  This is something you should do for
      yourself in the later problems on this problem set.
    \begin{enumerate}
    \item
\begin{problem}
        Draw a \textbf{snapshot} picture showing what is happening at the
        instant we care about.  (You can think of the radar station as a
        point.)  Label any distances you know.
      \end{problem}
      \pspace{\vfill}

      \begin{solution}
        We know that the plane is at an altitude of 1 mile.  After 6
        minutes, it has flown a horizontal distance of (500 miles per
        hour)$\left( \frac{1}{10} \text{ hour} \right)$ = 50 miles past the
        radar station.  So, here's a picture of the situation:
        \begin{center}
          \begin{tikzpicture}
            \graphinside
            \draw[blue] (station) -- (plane);
            \draw[dashed] (plane) -- node[right]{1 mi} (plane|-station);
            \draw[decorate, decoration={brace, mirror}] ($(station) - (0,
            3pt)$) -- node[below, yshift=-3pt]{50 mi} ($(plane |- station) - (0, 3pt)$);
          \end{tikzpicture}
        \end{center}
        (From this picture, we could also find the blue distance using the
        Pythagorean Theorem, but it's not necessary to do so at this
        stage.) 
      \end{solution}

    \item
\begin{problem}
        Draw another picture (the \textbf{generic} picture) showing the
        situation at an arbitrary time $t$.  On this picture, give
        variable names to quantities that are changing over time, and
        label any values that are constant.
      \end{problem}
      \pspace{\vfill}

      \begin{solution}
        The distance we labeled as 50 mi in
        {{{{\#2}}(b)}} changes as time goes
        on, so we should give that distance a variable name; let's call it
        $x$.  The distance between the radar station and plane is also
        changing; since that's what we're interested in, let's give that a
        name as well, like $z$.
        \begin{center}
          \begin{tikzpicture}
            \graphinside
            \draw[blue] (station) -- node[anchor=south]{$z$} (plane);
            \draw[dashed] (plane) -- node[right]{1 mi} (plane|-station);
            \draw[decorate, decoration={brace, mirror}] ($(station) - (0,

          \end{tikzpicture}
        \end{center}        
      \end{solution}

    \item
      \begin{problem}
        State your goal, together with what you are told in the problem.
        (For instance, ``Goal: Find $\frac{da}{dt}$.  Known: $\frac{db}{dt}
        = 2$.'')  If you use variables other than those you already labeled
        in {{{{\#2}}(b)}}, label them.

        \emph{In general, when you reach this step, it's important to think
          carefully about what the signs (positive or negative) of the rates
          you know are.}
      \end{problem}
      \pspace{\vfill}

      \begin{solution}
        Goal: Find $\frac{dz}{dt}$.  Known: $\frac{dx}{dt} = 500$ (this is
        positive because the distance $x$ is increasing: the plane is
        getting further from the radar station). 
      \end{solution}

      \pspace{\newpage}

    \item
\begin{problem}
        Write an equation relating the relevant variables.\footnote{Which
          are the relevant variables?  Well, if you know $\frac{db}{dt}$ and
          you want to know $\frac{da}{dt}$, then the relevant variables are
          $a$ and $b$.}
      \end{problem}
      \pspace{\stretch{0.5}}

      \begin{solution}
        We need to relate $z$ and $x$, and we can use our generic picture
        to do it.  By the Pythagorean Theorem, $x^2 + 1 = z^2$. 
      \end{solution}

    \item
      \begin{problem}
        Solve!  (You'll first to take the derivative of something you wrote in
        {{{{\#2}}(d)}}.)
      \end{problem}
      \pspace{\stretch{2}}

      \begin{solution}
        We differentiate both sides of the equation $x^2 + 1 = z^2$ with
        respect to $t$:
        \begin{align*}
          \frac{d}{dt} (x^2 + 1) &= \frac{d}{dt} (z^2) \\
          2x \frac{dx}{dt} &= 2z \frac{dz}{dt} \\
          \intertext{Now, plug in what we know and the snapshot
            information (we can get $z$ in the snapshot using the
            Pythagorean Theorem):}
          2 (50) (500) &= 2 \sqrt{50^2 + 1^2} \frac{dz}{dt}
        \end{align*}
        Solving, $\frac{dz}{dt} = \frac{25000}{\sqrt{2501}}$.  That is,
        the distance between the plane and the station is increasing at
        $\boxed{\frac{25000}{\sqrt{2501}} \text{ miles per hour}}$. 
      \end{solution}

\\

      \end{center}

    \begin{solution}
      We found in equation (3) above
      that $\frac{dV}{dt}=\frac{\pi}{4} h^2 \frac{dh}{dt}$.  If Valentina
      sips at a constant rate of 2 cm$^3$/second, then $\frac{dV}{dt} =
      -2$, and equation (3) becomes
      $-2 = \frac{\pi}{4} h^2 \frac{dh}{dt}$.  So, $\frac{dh}{dt} = -
      \frac{8}{\pi h^2}$.  As time goes on, $h$ decreases, so $-
      \frac{8}{\pi h^2}$ becomes more and more negative.  Therefore, the
      height \fbox{decreases more quickly} as time goes on.  (This makes
      sense intuitively; losing 1 cm$^3$ of liquid when the cone is very
      full shouldn't affect the height very much because the liquid at the
      top is very wide; on the other hand, losing 1 cm$^3$ of liquid when
      the cone is nearly empty will have a big effect on the height since
      the cone is so narrow at the bottom.)
    \end{solution}

  \end{enumerate}

\begin{problem}
        In economics, \textbf{Cobb-Douglas functions} can be used to model how
        a company's production amount ($P$) depends on the amount of labor
        involved ($L$) and the amount of capital invested ($K$). For a
        particular company, this relationship is
        \[ P = L^{2/3} K^{1/3}.\] Suppose the company's current labor force is
        $L = 1000$ workers, and the company is currently investing $K =
        \$64,000$ in capital to produce a particular product.
        Due to economic difficulty, the company is currently laying off
        workers at an instantaneous rate of 2 workers per week. If the
        company would like to keep its production level constant, at what
        instantaneous rate must it change its capital investment (with
        respect to time) right now?
    \end{problem}
    \pspace{\vfill}

    \begin{solution} % Jason Bland
        We are asked to find $\frac{dK}{dt}$, and we're told that
        $\frac{dL}{dt} = -2$ workers per week. So, this is a related
        rates question, and our next step should be to relate $K$ and $L$.
        Since the company would like to keep its production
            level constant, it wants $\frac{dP}{dt} = 0$.
        \begin{align*}
            L^{2/3} K^{1/3} &= P \\
            \intertext{Now, we'll differentiate both sides with respect to
                $t$:} \frac{d}{dt} \left( L^{2/3} K^{1/3} \right) &=
            \frac{dP}{dt} \\
            \frac{2}{3} L^{-1/3} \frac{dL}{dt} \cdot K^{1/3} + L^{2/3}
            \cdot \frac{1}{3} K^{-2/3} \frac{dK}{dt} &= 0 \\
            \intertext{Right now, $L = 1000$, $K = 64000$, and
                $\frac{dL}{dt} = -2$:} \frac{2}{3} (1000)^{-1/3} (-2)
            (64000)^{1/3} + (1000)^{2/3} \cdot \frac{1}{3}
            (64000)^{-2/3} \frac{dK}{dt} &= 0 \\
            \frac{2}{3} \cdot \frac{1}{10} \cdot (-2) \cdot (40) + (100)
            \cdot \frac{1}{3} \cdot \frac{1}{1600} \cdot \frac{dK}{dt}
            &= 0 \\
            -\frac{16}{3} + \frac{1}{48} \frac{dK}{dt} &= 0,
        \end{align*}
        so $\displaystyle \frac{dK}{dt} = \frac{48 \cdot 16}{3} = 256$.
        That is, the company must \\ \fbox{increase its capital investment at
            an instantaneous rate of \$256 per week}. 
    \end{solution}

 
\end{enumerate}
