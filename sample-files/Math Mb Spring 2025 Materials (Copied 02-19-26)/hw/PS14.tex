      x &= \boxed{\pm \sqrt{\frac{\pi}{2}}, \pm \sqrt{\frac{3 \pi}{2}},
        \pm \sqrt{\frac{5 \pi}{2}}}
    \end{align*}
  \end{solution}

  A ship is at sea and is heading to port $A$. The captain knows
  the distance from the ship to port $A$, the distance from his ship to
  port $B$, and the distance between the two ports. Due to weather
  conditions, the captain wants to charge course and head to port $B$.
  Through what angle must the ship turn?
  
  \textbf{Hint.} Recall the Law of Cosines: $a^2+b^2-2ab\cos C = c^2$. Your final answer will be in terms of $a$, $b$, and $c$.

  \begin{tikzpicture}
    \coordinate (C) at (0,0);
    \coordinate (B) at (3,1);
    \coordinate (A) at (4,-2);
    \filldraw[radius=2pt,fill=black]
      (C) circle node[left] {\includegraphics{images/boat.pdf}$C$}
      (B) circle node[right] {Port $B$}
      (A) circle node[right] {Port $A$};
    \draw (C) -- node[above] {$a$} (B) -- node[right] {$c$} (A) --
    node[below] {$b$} cycle;
    \draw pic[draw=black,angle radius=10] {angle=A--C--B};
  \end{tikzpicture}

  \pspace{\vfill\newpage}


  \begin{solution}
      This problem is essentially asking us to find angle $C$. Using the Law of Cosines and inverse trigonometric functions, we can do so as follows:

      \begin{align*}
          a^2+b^2-2ab\cos(C) &= c^2\\
          a^2+b^2-c^2 &= 2ab\cos(C)\\
          \frac{a^2+b^2-c^2}{2ab} &= \cos(C) \\
           \arccos\left(\frac{a^2+b^2-c^2}{2ab}\right) &= C 
      \end{align*}

      Therefore, $\boxed{C=\arccos\left(\frac{a^2+b^2-c^2}{2ab}\right)}$.

      Notice that if $c^2=a^2+b^2$, then $$C=\arccos(0)=\frac{\pi}{2}=90^{\circ}$$ This makes sense, because if $c^2=a^2+b^2$, then triangle $ABC$ must be a right triangle with side $c$ as its hypotenuse, making angle $C$ a right angle. 
  \end{solution}

  A population of deer in a forest displays regular fluctuations in
  size. Scientists have chosen to model the population size as a
  sinusoidal function of time.  At its height the deer population is 7000,
  while at its low it is 2000. The time between highs and lows is 6
  months. The population at time $t=0$ is 4500 and decreasing.
  \begin{enumerate}

  \item
    \begin{problem}[\vfill\newpage]
      Model the deer population as a function of time $t$ in
      months. Please include a sketch along with a formula. 
    \end{problem}

    \begin{solution}
      Since the time between highs and lows is 6 months, the period must
      be 12 months.  Also, the balance value is half-way between $2000$
      and $7000$, which is $4500$.  Here's a sketch of the deer population
      as a function of $t$:
      \begin{center}
        \begin{tikzpicture}
          \begin{axis}[xtick={3, 6, ..., 24}, ytick={2000,7000}, extra y
            ticks={4500}, xlabel=$t$, ylabel=population, width=3.5in,
            height=2in]
            \addplot+[domain=0:26]{-2500*sin(deg(pi/6*x))+4500};
          \end{axis}
        \end{tikzpicture}
      \end{center}
      The sinusoid's amplitude is 2500.  Since the graph starts at its
      balance value and decreases, we can get it by starting with $- \sin
      t$, stretching horizontally by a factor of $\frac{12}{2 \pi}$ (to
      get the right period), stretching vertically by $2500$ (to get the
      right amplitude), and shifting vertically by $4500$ (to get the
      right balance value).  So, our formula is $\boxed{P(t)=-2500 \sin
        \left(\frac{\pi}{6} t \right)+ 4500}$.
    \end{solution}

  \item
    \begin{problem}[\stretch{0.5}]
      If $t=0$ is now, what is the deer population 3 months from now?
    \end{problem}

    \begin{solution}
      We can plug in $t = 3$ to get $P(3) = \boxed{2000}$ (or we can just
      look at the graph).
    \end{solution}

  \item
    \begin{problem}[\vfill]
      When are the first three times in the future that the deer

      We'd like the 3 smallest positive values of $t$ such that

        -2500 \sin \left( \frac{\pi}{6} t \right)  + 4500 &= 3000 \\

        \sin \left( \frac{\pi}{6} t \right) &= \frac{-1500}{-2500} =
        \frac{3}{5} \\
        \shortintertext{Let's substitute $u = \frac{\pi}{6} t$:}
        \sin (u) &= \frac{3}{5}
      \end{align*}
      Positive values of $t$ correspond to positive values of $u$, so we'd
      like to find the 3 smallest positive solutions of $\sin (u) =
      \frac{3}{5}$.  One solution is $\arcsin \frac{3}{5}$; let's call
      this $A$:

        \begin{tikzangle}[angle=rad(asin(3/5)), label=$A$, func=sin,

        \end{tikzangle}
      \end{center}
      Then, here are the 3 smallest positive solutions of $\sin(u) =
      \frac{3}{5}$:

        -2500 \sin \left( \frac{\pi}{6} t \right)  + 4500 &= 6500 \\
        - 2500 \sin \left( \frac{\pi}{6} t \right) &= 2000 \\
        \sin \left( \frac{\pi}{6} t \right) &= \frac{2000}{-2500} =
        -\frac{4}{5} \\
        \shortintertext{Let's substitute $u = \frac{\pi}{6} t$:}
        \sin (u) &= -\frac{4}{5}
      \end{align*}
      Positive values of $t$ correspond to positive values of $u$, so we'd
      like to find the 3 smallest positive solutions of $\sin (u) =
      -\frac{3}{5}$.  One solution is $\arcsin \left( -\frac{4}{5}
      \right)$, but I find it confusing to work in terms of this because
      it's negative; instead, I'll write the solutions in terms of $B =
      \arcsin \frac{4}{5}$:
      \begin{center}
        \begin{tikzangle}[angle=rad(asin(3/5)), label=$B$, func=sin,
          functick=$\frac{4}{5}$, color=blue] 
          \draw[blue] (axis cs: 1.5, 0) node[right] {$B = \arcsin
            \frac{4}{5}$}; 
        \end{tikzangle}
      \end{center}
      Then, here are the 3 smallest positive solutions of $\sin(u) =
      -\frac{4}{5}$:
      \begin{center}
        \begin{tabular}{ccc}
          \begin{tikzangle}[angle=pi+rad(asin(4/5)), label=$u$, func=sin,
            functick=$-\frac{4}{5}$, secondangle=rad(asin(4/5)),
            secondlabel=$B$, seconddistance=0.24]
            \draw[funcline] (axis cs: 0.6, 0.8) -- (axis cs: -0.6, 0.8);
            \draw (axis cs: 0, 0.8) node[funcxtick]{$\frac{4}{5}$}; 
          \end{tikzangle} &
          \begin{tikzangle}[angle=2*pi-rad(asin(4/5)), label=$u$, func=sin,
            functick=$-\frac{4}{5}$, secondangle=rad(asin(4/5)),
            secondlabel=$B$, seconddistance=0.24]
            \draw[funcline] (axis cs: 0.6, 0.8) -- (axis cs: -0.6, 0.8);
            \draw (axis cs: 0, 0.8) node[funcxtick]{$\frac{4}{5}$}; 
          \end{tikzangle} &
          \begin{tikzangle}[angle=3*pi+rad(asin(4/5)), label=$u$, func=sin,
            functick=$-\frac{4}{5}$, secondangle=rad(asin(4/5)),
            secondlabel=$B$, seconddistance=0.24]
            \draw[funcline] (axis cs: 0.6, 0.8) -- (axis cs: -0.6, 0.8);
            \draw (axis cs: 0, 0.8) node[funcxtick]{$\frac{4}{5}$}; 
          \end{tikzangle} \\
          $u = \pi + B$ & $u = 2\pi - B$ & $u = 3 \pi + B$
        \end{tabular}
      \end{center}
      So, the 3 smallest solutions are $u = \pi + \arcsin \frac{4}{5}, 2
      \pi - \arcsin \frac{4}{5}, 3 \pi + \arcsin \frac{4}{5}$.  Remember
      that $u = \frac{\pi}{6} t$, so $t = \frac{6}{\pi} u = \boxed{
        \frac{6}{\pi} \left( \pi + \arcsin \frac{4}{5} \right),
        \frac{6}{\pi} \left( 2 \pi - \arcsin \frac{4}{5} \right),
        \frac{6}{\pi} \left( 3 \pi + \arcsin \frac{4}{5} \right)}$.  These
      are $\boxed{\approx 7.771, 10.229, 19.771}$.
    \end{solution}

  \end{enumerate}
