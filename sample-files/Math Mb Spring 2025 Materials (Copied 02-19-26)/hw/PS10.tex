\begin{problem}
A mass attached to a spring will oscillate back and forth when set in motion. This is called \emph{simple harmonic motion}, and it turns out that the position of the mass can be described \emph{exactly} by a sinusoidal function. (Mass-spring systems are used to model more complex physical and biological systems.)

\begin{center}
\includegraphics[width=3in]{images/spring-diagram.pdf}
\end{center}

A mass attached to a spring oscillates back and forth between $x=+3$ cm and $x=-3$ cm with a period of $10$ s. At $t=0$, the mass is at $x=0$ and is traveling in the negative direction. Find a sinusoidal function of time that describes the mass's position. (It may help to sketch the graph first.)
    \end{problem}
    \pspace{\vfill}

\begin{solution}
The position-time graph for the mass looks like this.

\begin{center}
    \begin{tikzpicture}
        \begin{axis}[
        xlabel=$t$ (s),
        ylabel=$x$ (cm),
        ytick={-3,0,3},
        minor y tick num=2,
        minor x tick num=1,
        grid=both,
        ]
            \addplot[domain=0:20] {-3*sin(deg(x*pi/5))};
        \end{axis}
    \end{tikzpicture}
\end{center}
\end{solution}

The amplitude is $3$, the midline value is $0$, and the period is $10$. At $t=0$, the wave is decreasing through the midline. A function that describes this behavior could be \fbox{$f(t)=-3\sin(\frac{\pi}{5}t)$.}

    ]

    \nextgroupplot[title={(a) $y=\sin^2(x)$}]
    \addplot[very thick] {sin(deg(x))^2};


    \nextgroupplot[title={(b) $y=\cos^2(x)$}]
    \addplot[very thick] {cos(deg(x))^2};


    \end{groupplot}
\end{tikzpicture}

    \end{center}
\end{problem}

\begin{enumerate}
    \item 
    \begin{problem}
    Given the graph of $y=\sin^2x$ shown, rewrite $\sin^2x$ in sinusoidal form.
    \end{problem}
    
    \vspace{16pt}
    $\sin^2x=\solonly{\boxed{-\frac{1}{2}\cos(2x)+\frac{1}{2}}}$ 
    \vspace{16pt}
    
    \item
    \begin{problem}
    Given the graph of $y=\cos^2x$ shown, rewrite $\cos^2x$ in sinusoidal form.
    \end{problem}
    
    \vspace{16pt}
    $\cos^2x=\solonly{\boxed{\frac{1}{2}\cos(2x)+\frac{1}{2}}}$ 
    \vspace{16pt}

\end{enumerate}
\probonly{\emph{These trigonometric identities are known as the power-reducing formulas.}}

  The average rental price for a two-bedroom apartment in Malden was \$800
  in 1990 and \$1000 in 2000. The price was increasing over the entire
  decade. We want to model the price of a two-bedroom apartment in Malden
  as a function of time.


  \begin{enumerate}

  \item Suppose we let $t=0$ correspond to the year 1990 and measure time
    in years. Help your friends find formulas for the models they propose. Accompany each
    function formula with a sketch.
    \probonly{}
      \begin{enumerate}
      \item
        \begin{problem}[\vfill]
          Alex thinks that prices are increasing at a constant rate, so he wants to
    model the price with a linear function, $L(t)$.
        \end{problem}

        \begin{solution}
          We're looking for the equation of the line through $(0, 800)$
          and $(10, 1000)$.  This line has slope $20$ and $y$-intercept
          800, so its equation is \fbox{$L(t) = 20t + 800 $}.
        \end{solution}

      \item
        \begin{problem}[\vfill]
            Jamey thinks that the percent change in price is constant, so he
    wants to model the price with an exponential function, $E(t)$.
        \end{problem}

        \begin{solution}
          The price has grown by 25\% in 10 years, so an exponential model
          is \fbox{$E(t) = 800\cdot1.25^{t/10} $}.
        \end{solution}


      \item
        \begin{problem}[\vfill]
            Mike, an optimist who loves trigonometry, thinks that price is a
    sinusoidal function of time. He thinks that \$800 is an all-time low
    and \$1000 is an all-time high. He wants to model the price with a sinusoidal
    function, $T(t)$.
        \end{problem}

        \begin{solution}
          The graph of the function that Mike is imagining looks like
          this:
          \begin{center}
            \begin{tikzpicture}
              \begin{axis}[ymin=0]
                \addplot+[domain=0:40] {900-100*cos(deg(pi/10*x))};
              \end{axis}
            \end{tikzpicture}
          \end{center}
          This sinusoid has amplitude $100$, balance value $900$, and
          period 20.  Since it has a minimum at $t = 0$, it's easiest to
          model it by starting with $-\cos t$: if we stretch this
          vertically by 100, horizontally by $\frac{20}{2\pi}$, and then
          move it up by $900$, we'll get Mike's graph.  This gives us the
          formula $\boxed{ T(t) = -100\cos\left(\frac{\pi}{10}t\right) +
            900}$.
        \end{solution}

      \end{enumerate}      
      \probonly{}

    \probonly{\emph{You can graph your formulas
        with a calculator to check whether they match your graphs.}}


        \pspace{\newpage}
  \item
    \begin{problem}[\stretch{0.8}]
      Which model predicts the highest price for the year 2003? Which
      model predicts the lowest price for the year 2003?
    \end{problem}

    \begin{solution}
      Both the linear and exponential models continue to increase after
      the year 2000, so \fbox{Mike's sinusoidal model predicts the lowest
        price}.  The exponential function will grow faster than the linear
      one after 2000, so \fbox{Jamey's exponential function predicts the
        highest price}.  
    \end{solution}

  \item Alex, Jamey, and Mike are combing the newspapers for information
    that might lend credence to one model over the others. Which model ---
    the linear, exponential, or sinusoidal --- is best supported by each
    of the following statements?  (Choose one model for each
    statement.)
    \begin{enumerate}

    \item
      \begin{problem}[\stretch{0.4}]
        ``Prices in Malden have been growing at an increasing rate over
        the past decade.''
      \end{problem}

      \begin{solution}
        This supports the exponential model; graphically, it says that the
        price graph is concave up.
      \end{solution}

    \item
      \begin{problem}[\stretch{0.4}]
        ``In the early 1990s prices in Malden were increasing at an
        increasing rate. After 1995 the rate of increase began to drop
        off.''
      \end{problem}

      \begin{solution}
        This supports the sinusoidal model; graphically, it says that the
        price graph is concave up on $(0, 5)$ and concave down on $(5,
        10)$. 
      \end{solution}

    \item
      \begin{problem}[\stretch{0.4}]
        ``Prices of apartments in Malden have been increasing very
        steadily over the past decade.''
      \end{problem}

      \begin{solution}
        This supports the linear model. 
      \end{solution}

    \end{enumerate}

  \end{enumerate}

  \begin{problem}
    Recall from Math Ma that the graph of a polynomial function can have at most $n$ $x$-intercepts and at most $n-1$ turning points, where $n$ is the degree of the polynomial. The end behavior (as $x\to\pm\infty$) depends on the degree and the sign of the leading coefficient but is always either $y\to \infty$ or $y\to-\infty$.
    
    Explain how the graph of $f(x) = \sin x$ indicates that it is not a polynomial function. Refer to the number of $x$-intercepts, the number of
    turning points, and the end behavior.
  \end{problem}
  
    \begin{solution}
     The graph of $y=\sin(x)$ has
    infinitely many $x$-intercepts and turning points, and $\displaystyle \lim_{x
      \to \infty} \sin x$ and $\displaystyle \lim_{x \to -\infty} \sin x$ do not
    exist because both ``ends'' of the graph oscillate between $-1$ and
    $1$ forever.  So, $\sin x$ cannot be a polynomial. 
  \end{solution}

\begin{problem}
    Next class, we will investigate derivatives of trigonometric functions. Let $f(x)=\sin x$ and $g(x)=\cos x$.
\end{problem}

\begin{enumerate}
    \item 
    \begin{problem}
    Fill in the limit definition of $f'(0)$. $f'(0) = \displaystyle
      \lim_{h \to 0} \fitb[\vphantom{\dfrac{0}{0}}]{3in}{\frac{\sin(h)}{h}}$
      \end{problem}

    \item 
    \begin{problem}
    Use a calculator to find approximations of $f'(0)$. (Plug in values of $h$ close to $0$.) Come up with a guess for the exact value.
      \end{problem}
      \pspace{\vfill}

    \begin{solution}
        Here is a table of a few approximations.
        
\begin{tabular}{c|c}
$h$     & $\frac{\sin h}{h}$ \\ \hline
$-0.1$  & $0.998$           \\
$-0.01$ & $0.999$           \\
$0$     & undefined          \\
$0.01$  & $0.999$           \\
$0.1$   & $0.998$          
\end{tabular}

Based on these values, it seems like $f'(0)=1$.
    \end{solution}

        \item 
    \begin{problem}
    Fill in the limit definition of $g'(0)$. $g'(0) = \displaystyle
      \lim_{h \to 0} \fitb[\vphantom{\dfrac{0}{0}}]{3in}{\frac{\cos(h)-1}{h}}$
      \end{problem}

    \item 
    \begin{problem}
    Use a calculator to find approximations of $g'(0)$. (Plug in values of $h$ close to $0$.) Come up with a guess for the exact value.
      \end{problem}
      \pspace{\vfill}

    \begin{solution}
        Here is a table of a few approximations.

        \begin{tabular}{c|c}
$h$     & $\frac{\cos h-1}{h}$ \\ \hline
$-0.1$  & $0.050$             \\
$-0.01$ & $0.005$             \\
$0$     & undefined            \\
$0.01$  & $-0.005$            \\
$0.1$   & $-0.050$           
\end{tabular}

Based on these values, it seems like $g'(0)=0$.
    \end{solution}

    \item 
    \begin{problem}
        Are your guesses for $f'(0)$ and $g'(0)$ consistent with the graphs of sine and cosine? Explain briefly.
    \end{problem}
    \pspace{\vfill}

    \begin{solution}
        The graph of $y=\sin(x)$ is increasing at $x=0$, so $f'(0)=1$ is consistent with the graph of sine. The graph of $y=\cos(x)$ has a turning point and horizontal tangent at $x=0$, so $g'(0)=0$ is consistent with the graph of cosine.
    \end{solution}
\end{enumerate}

\end{enumerate}
