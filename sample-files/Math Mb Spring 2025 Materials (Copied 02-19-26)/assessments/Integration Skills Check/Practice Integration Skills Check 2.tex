\item Find the following basic \textbf{indefinite} integrals.

\begin{enumerate}
    \item $\displaystyle \int x^n \;dx$ (for $n \neq -1$)
    \vfill 
    \answerbox[\solonly{$\frac{1}{n+1}x^{n+1}+C$}]{.5 in}{5.5 in}
    
    \item $\displaystyle \int \sin(x) \;dx$
    \vfill 
    \answerbox[\solonly{$-\cos(x)+C$}]{.5 in}{5.5 in}
    
    \item $\displaystyle \int e^x \;dx$
    \vfill 
    \answerbox[\solonly{$e^x+C$}]{.5 in}{5.5 in}

    \item $\displaystyle \int \frac{1}{1+x^2} \;dx$
    \vfill 
    \answerbox[\solonly{$\arctan(x) +C$}]{.5 in}{5.5 in}
    
    \item $\displaystyle \int \frac{1}{x}\;dx $
    \vfill 
    \answerbox[\solonly{$\ln\!|x|+C$}]{.5 in}{5.5 in}
    
    \item $\displaystyle \int \cos(x)\;dx$
    \vfill 
    \answerbox[\solonly{$\sin(x)+C$}]{.5 in}{5.5 in}

\end{enumerate}

\newpage 



\item Find each integral. You may use any method. \textbf{Fully simplify your answers to all definite integrals.}
\begin{enumerate}

    \item $\displaystyle \int_0^1 x^2 \;dx$
    \vfill 
    \answerbox[\solonly{$\frac{1}{3}$}]{.5 in}{5.5 in}
    
    \item $\displaystyle \int_0^{\frac{3\pi}{2}} x \sin(x) \;dx$
    \vfill 
    \answerbox[\solonly{$-1$}]{.5 in}{5.5 in}
    
    \item $ \displaystyle \int \tan(t) \;dt$
    \vfill 
    \answerbox[\solonly{$-\ln\!|\!\cos(t)|+C$}]{.5 in}{5.5 in}

    \newpage 
    
    \item $\displaystyle \int (4x^3 - 3x^2 +x - 7) \;dx$
    \vfill 
    \answerbox[\solonly{$x^4-x^3+\frac{1}{2}x^2-7x+C$}]{.5 in}{5.5 in}
    
    \item $ \displaystyle \int_2^4 |t-3| \; dt$
    \vfill 
    \answerbox[\solonly{$1$}]{.5 in}{5.5 in}
    
    \item $\displaystyle \int \frac{x^3 - \sqrt{x}}{x} \;dx$
    \vfill 
    \answerbox[\solonly{$\frac{1}{3}x^3-2\sqrt{x}+C$}]{.5 in}{5.5 in}

    \newpage 
    
    \item $\displaystyle \int_{1}^{2} 4x e^{x^2}\;dx$. You do not need to simplify your answer.
    \vfill 
    \answerbox[\solonly{$2e^4-2e$}]{.5 in}{5.5 in}
    
    \item $\displaystyle \int e^t \cdot t\;dt$
    \vfill 
    \answerbox[\solonly{$te^t-e^t+C$}]{.5 in}{5.5 in}


    \item $\displaystyle \int \dfrac{x}{x-4} \;dx$
    \vfill 
    \answerbox[\solonly{$x+4\ln\!|x-4|+C$}]{.5 in}{5.5 in}

\end{enumerate}

\item We want to approximate the area of the region $\mathcal{R}$ shown below using a right Riemann sum with 10 equal sub-intervals, $R_{10}$. $\mathcal{R}$ is bounded by the curves $y=f(x)$, $y=0$, $x=1$, and $x=4$, where $f(x)=x^2$.

    \begin{tikzpicture} 
      \begin{axis}[stack plots=y, cycle list={black}, xmin=0, xmax=4.25, xlabel=$x$, ylabel=$y$,
        width=2.5in, height=2.5in, clip=false, ytick=\empty] 
        \addplot+[domain=1:4]{0} node [right] {\;} ; 
        \addplot+[domain=1:4, fill=lightgray, fill opacity=0.2] 
        {x^2} \closedcycle; 
        \node [left] at (axis cs: 2, 5) {$y = f(x)$};
        \node at (axis cs: 3, 4) {$\mathcal{R}$};
      \end{axis} 
    \end{tikzpicture}
\begin{enumerate}

    \item Write down the value of $\Delta x$.
    \vfill

    \answerbox[$\Delta x =\solonly{\frac{3}{10}}$]{.5 in}{5.5 in}
    
    \item Write down a formula for $x_k$, the right endpoint of the $k$th sub-interval.
    \vfill
    
    \answerbox[$x_k=\solonly{1+\frac{3}{10}k}$]{.5 in}{5.5 in}
    
    \item Write down a formula for $R_{10}$ using sigma notation. 
    \vfill
    
    \answerbox[$R_{10}=\solonly{\displaystyle \sum_{k=1}^{10} f(x_k)\Delta x}$ \solonly{ or $R_{10}=\displaystyle \sum_{k=1}^{10} \tfrac{3}{10}(1+\tfrac{3}{10}k)^2$}]{.75 in}{5.5 in}
\end{enumerate}
    
    
\end{enumerate}
