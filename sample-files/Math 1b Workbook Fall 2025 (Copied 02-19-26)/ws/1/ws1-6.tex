\section{Numerical Integration} \label{ws1-6}
\begin{framed}
\textbf{Lesson Goals:}
\begin{itemize}[parsep=0pt]
\item You should be able to approximate the value of a definite integral by finding, left-hand, right-hand, trapezoidal, and midpoint/midpoint tangent Riemann sums with $n$ slices, $L_n$, $R_n$, $T_n$, and $M_n$. You should also be able to draw these approximations, and determine when these approximations are overestimates or underestimates of the integral's value.
\item You should be able to explain the relationships between $L_n$, $R_n$, $T_n$, and $M_n$.
\end{itemize}
\end{framed}

\note{The theme for the day is: we know how to approximate, and we even saw at the end of last class how many $n$ we 
need to get the accuracy we desire, but this $n$ is sometimes very high.  We wonder if it's possible to do better 
with a different approximation.  Some students will have seen at least $T_n$ before, and $M_n$ is sort of natural

\note{Introduce the midpoint rule approximation $M_{n}$.  It's helpful to
point out how the trapezoidal rule involves averaging $f(x_{k-1})$ and $f(x_k)$,
whereas the midpoint rule involves averaging $x_{k-1}$ and $x_{k}$.}


\item 
  \begin{problem}[3in]
    Consider the integral $\displaystyle \int_{-\pi/2}^{\pi/2} \cos(x)~dx$.  Write
    out the midpoint approximation $M_{4}$, showing  
    all terms in the sum.  There is no need to simplify your answer.
  \end{problem}

  \begin{solution}
    Computing $M_4$ requires us to slice $[-\pi/2, \pi/2]$ into four subintervals
    of length $\Delta x = \pi/4$.  Let
    $$
      x_0 = -\frac{\pi}{2}, \qquad
      x_1 = -\frac{\pi}{4}, \qquad
      x_2 = 0, \qquad
      x_3 = \frac{\pi}{4}, \qquad
      x_4 = \frac{\pi}{2},
    $$
    and let $f(x) = \cos(x)$.    
    

    The midpoint sum $M_{4}$ does not involve computing $f(x) = \cos(x)$ at the
    endpoints of the subintervals---only at the midpoints of the subintervals:
    $$
      \begin{aligned}
      M_{4} & = f \left( \frac{x_0 + x_1}{2} \right) \Delta x +
                f \left( \frac{x_1 + x_2}{2} \right) \Delta x +
                f \left( \frac{x_2 + x_3}{2} \right) \Delta x +
                f \left( \frac{x_3 + x_4}{2} \right) \Delta x \\
            & = [ \cos (-3 \pi/8)  + \cos(- \pi/8)  +
                \cos(\pi/8)  + \cos(3 \pi/8) ] \Delta x.
      \end{aligned}
    $$

    The figure below indicates that $T_{4}$ is an {\em underestimate} of the
    true integral.  The reason for this is that $f$ is concave down throughout
    the domain of integration---you can check this by looking at the sign of
    the second derivative of $f$.  The ``upper" edge of each trapezoid in the
    figure is a secant line segment and, when the graph of a function is concave
    down, secant line segments lie below the graph.

    \begin{center}
      \includegraphics[width=2in]{ws/images/cosine_T4_approximation.png}
    \end{center}


  \end{solution}

   \item \begin{problem}
 Suppose we want to approximate
  $\displaystyle{\int_a^b f(x) ~ dx}$ using the \textit{trapezoidal rule} $T_n$.
  We partition $[a,b]$ into $n$ equal pieces each
  of length $\Delta x$, let $x_k = a + k\Delta x$
  for $k = 0, 1, \dots, n$, and draw $n$ trapezoids by connecting the points $(x_k, f(x_k))$ and $(x_{k-1}, f(x_{k-1}))$ with a straight line for each $k$.
  Use a picture of a single slice to explain geometrically why the trapezoidal rule is the average of the left and right-hand approximations. That is, $T_n=\frac{L_n + R_n}{2}$. %\textit{Hint: notice that the area of a trapezoid with parallel bases $b_1$ and $b_2$ and height $h$ is given by $\frac{1}{2}(b_1+b_2)h$}.
  \end{problem}
  
  
  \begin{solution}
  Consider one subinterval, on which we look at a left-hand rectangle, a right-hand rectangle, and a trapezoid. 
    \begin{center}
  \includegraphics{ws/images/trapezoid-geometric-average}
  \end{center}
  
  The left-hand rectangle is AEFD, while the right-hand rectangle is ABCD. Notice that the area of trapezoid AECD is the area of rectangle ABCD plus triangle BEC. Also notice that the area of trapezoid AECD is the area of rectangle AEFD minus triangle EFC. Since triangles BEC and EFC have the same area, the trapezoid's area is exactly between the left and right rectangles, i.e., the average of the two.
  
  Since this geometric argument will hold true on any subinterval, the trapezoidal rule gives the average of the left and right-hand approximations.
  \end{solution}

\item
    Let $I=\displaystyle\int_0^{10} e^{-x^2}$. This integral does not have an elementary antiderivative. Using $n=10$ slices, we can approximate the value of $I$. Doing so, we see that $$R_{10}=0.38631860$$
        $$L_{10}=1.386318$$
    \begin{enumerate}
        \item\begin{problem}[\vfill] Write an expression that gives the amount of error in approximating $I$ with $L_{10}$.\end{problem}

       \begin{solution}
           $I-L_{10}$ gives the difference between the integral and our approximation. In order to focus solely on the size of the error, we will express the error as $|I-L_{10}|.$ 
       \end{solution}
        \item \begin{problem}[\vfill]
            Explain why we cannot find an exact value for this expression.
        \end{problem}
        
        \begin{solution}
            We cannot find the exact error because we do not know what $I$ is. Therefore we can't know exactly how far off our approximation is.
        \end{solution}
            
       
        \item \begin{problem}[\vfill]
            Can you find an upper bound of the error in approximating $I$ with $L_{10}$?
        \end{problem}
        
        \begin{solution}
            $L_{10}-R_{10}\approx1$ is an upper bound of the error because we know $L_{10}$ is an over-approximation and $R_{10}$ is an under-approximation since the function is decreasing on the interval. Thus, the error must be less than their difference.
        \end{solution}
        
        \item \begin{problem}[\vfill]
            How could you improve your approximation with the given information?
        \end{problem}
        
       \begin{solution}
           We could average $R_{10}$ and $L_{10}$ to approximate $I$ with $T_{10}$ which in general should be a better approximation.
       \end{solution}
    \end{enumerate}





\end{enumerate}
