\section{Improper Integrals I} \label{ws1-9}
\begin{framed}
\textbf{Lesson Goals:}
\begin{itemize}
\item You should be able to determine if an improper integral of a bounded function converges or diverges, by converting it to a limit of a proper integral and evaluating.
\item You should be able to determine when a $p$-integral converges or diverges.
\item You should be able to determine if an improper integral of a positive bounded function converges or diverges using the Comparison Theorem.
\item You should be able to articulate the geometric intuition behind the Comparison Theorem, as well as understand the conditions and conclusions of the theorem.
\item You should be able to reason abstractly about the convergence behavior of improper integrals.
\end{itemize}
\end{framed}

\begin{enumerate}
  \item 
  \begin{enumerate}
    \item
    \begin{problem}[2.5in]
    Does $\displaystyle \int_1^\infty \frac{1}{x^2}~dx$ converge or
    diverge?  If it converges, evaluate it.
    \end{problem}

    \begin{solution}
    We know that $\displaystyle \int_1^\infty \frac{1}{x^2}~dx$ really
    means $\displaystyle \lim_{b \to \infty} \int_1^b \frac{1}{x^2}~dx$.
    So, 
    \begin{align*}
    \int_1^\infty \frac{1}{x^2}~dx & = \lim_{b \to \infty} \int_1^b
    x^{-2}~dx \\
    & = \lim_{b \to \infty} \left( -x^{-1} \Big|_1^b
    \right) \\
    & = \lim_{b \to \infty} \left( -\frac{1}{b} + 1 \right)
%        & = \lim_{b \to \infty} \left( 1 - \frac{1}{b} \right) \\
\end{align*}
As $b \to \infty$, $\frac{1}{b} \to 0$, so this limit is equal to
\fbox{1}.  (Since the limit exists, the improper integral
converges.)
\end{solution}

\item
\begin{problem}[2.5in]
Does $\displaystyle \int_1^\infty \frac{1}{x}~dx$ converge or
diverge?  If it converges, evaluate it.
\end{problem}

\begin{solution}
We know that $\displaystyle \int_1^\infty \frac{1}{x}~dx$ really
means $\displaystyle \lim_{b \to \infty} \int_1^b \frac{1}{x}~dx$, so
\begin{align*}
\int_1^\infty \frac{1}{x}~dx & = \lim_{b \to \infty}
\left( \ln |x| \Big|_1^b \right) \\
& = \lim_{b \to \infty} \ln |b| \textrm{ since } \ln 1 = 0
\end{align*}
But we know that $\displaystyle \lim_{b \to \infty} \ln |b| =
\infty$, which is a form of diverging, so the improper integral
$\displaystyle \int_1^\infty \frac{1}{x}~dx$ \fbox{diverges}.
\end{solution}
\end{enumerate}

\begin{enumerate}
  \item
  \begin{problem}[2.3in]
  $\displaystyle \int_1^\infty \frac{1}{x^3}~dx$?
  \end{problem}

  \begin{solution}
  Let's graph $\frac{1}{x^2}$ (the solid curve) and $\frac{1}{x^3}$
  (the dashed curve):
  \begin{center}
  \includegraphics[width=2in]{images/improper-integrals-comparison-1-1}
  \end{center}
  By {{\#1}(a)}, we know that $\displaystyle \int_1^\infty
  \frac{1}{x^2}~dx = 1$.  Graphically, we interpret this as the area
  under the curve $y = \frac{1}{x^2}$ to the right of $x = 1$.  It
  is apparent from our picture that the area under $y =
  \frac{1}{x^3}$ to the right of $x = 1$ should be less than that,
  so we expect the integral $\displaystyle \int_1^\infty
  \frac{1}{x^3}~dx$ to converge.


  We can use the Comparison Theorem to justify our
  guess.\footnote{Another way to show that the improper integral
  converges is just to compute it as a limit of definite
  integrals, as we did in {\#1}.}  For $x \geq
  1$, $0 < \frac{1}{x^3} \leq \frac{1}{x^2}$ because $x^3 \geq x^2$.
  We showed in {{\#1}(a)} that $\displaystyle
  \int_1^\infty \frac{1}{x^2}~dx$ converges, so the Comparison
  Theorem says that $\displaystyle \int_1^\infty \frac{1}{x^3}~dx$
  must converge as well.
  \end{solution}

  \item
  \begin{problem}[2.3in]
  $\displaystyle \int_1^\infty \frac{1}{x^{1/2}}~dx$?
  \end{problem}

  \begin{solution}
  Let's graph $\frac{1}{x}$ (the solid curve) and
  $\frac{1}{x^{1/2}}$ (the dashed curve):
  \begin{center}
  \includegraphics[width=2in]{images/improper-integrals-comparison-1-2}
  \end{center}
  Since the graph of $\frac{1}{x^{1/2}}$ is higher than the graph of
  $\frac{1}{x}$ when $x \geq 1$, the area under $\frac{1}{x^{1/2}}$
  to the right of $x = 1$ should be at least as big as the area
  under $\frac{1}{x}$ to the right of $x = 1$.  The area under
  $\frac{1}{x}$ to the right of $x = 1$ is infinite (from
  {{\#1}(b)}), so $\displaystyle \int_1^\infty
  \frac{1}{x^{1/2}}~dx$ should diverge.

  To justify this mathematically, we use the Comparison Theorem. For
  $x \geq 1$, $0 < \frac{1}{x} \leq \frac{1}{x^{1/2}}$.  We know by
  {{\#1}(b)} that $\displaystyle \int_1^\infty
  \frac{1}{x}~dx$ diverges, so the Comparison Theorem says that
  $\displaystyle \int_1^\infty \frac{1}{x^{1/2}}~dx$ diverges as
  well. 
  \end{solution}

  \item
  \begin{problem}[2.3in]
  $\displaystyle \int_1^\infty \frac{1}{x^{3/2}}~dx$?
  \end{problem}

  \begin{solution}
  The graph of $\frac{1}{x^{3/2}}$ (dashed) lies between the graphs
  of $\frac{1}{x}$ (thin solid) and $\frac{1}{x^2}$ (thick solid):
  \begin{center}
  \includegraphics[width=2in]{images/improper-integrals-comparison-1-3}
  \end{center}
  Thinking in terms of areas, we can see that $\displaystyle
  \int_1^\infty \frac{1}{x^2}~dx \leq \int_1^\infty
  \frac{1}{x^{3/2}}~dx \leq \int_1^\infty \frac{1}{x}~dx$.  Using
  our result from {\#1}, this tells us that
  $\displaystyle 1 \leq \int_1^\infty \frac{1}{x^{3/2}}~dx \leq
  \infty$.  Unfortunately, that doesn't give us enough information
  to determine whether $\displaystyle \int_1^\infty
  \frac{1}{x^{3/2}}~dx$ converges.\footnote{It could be a finite
  number (in which case it converges) or it could be infinite (in
  which case it diverges).}

  In this case, comparing to $\frac{1}{x}$ and $\frac{1}{x^2}$
  doesn't help, so we'll just evaluate the improper integral using
  the definition:
  \begin{align*}
  \int_1^\infty \frac{1}{x^{3/2}}~dx & = \lim_{b \to \infty}
  \int_1^b \frac{1}{x^{3/2}}~dx \\
  & = \lim_{b \to \infty} \left( \left. -2 x^{-1/2} \right|_1^b
  \right) \\
  & = \lim_{b \to \infty} -2b^{-1/2} + 2 \\
  & = \lim_{b \to \infty} 2 - \frac{2}{\sqrt{b}} \\
  & = 2
  \end{align*}
  So, the improper integral converges to 2.
  \end{solution}

\end{enumerate}

\begin{framed}
\centerline{\textbf{The Comparison Theorem}}
Let $f(x)$ and $g(x)$ be two continuous functions on the interval $[a, \infty)$ such that $$0\leq f(x) \leq g(x)$$ for all $x$, \textit{eventually}. Then:
\begin{itemize}
\item If $\displaystyle \int_a^{\infty} g(x)\ dx$ \uline{converges}, then $\displaystyle \int_a^{\infty} f(x)\ dx$ \probonly{\underline{\hspace{1.5in}}} \solonly{\uline{converges as well}}. 
\item If $\displaystyle \int_a^{\infty} f(x)\ dx$ \uline{diverges}, then $\displaystyle \int_a^{\infty} g(x)\ dx$ \probonly{\underline{\hspace{1.5in}}} \solonly{\uline{diverges as well}}. 
\end{itemize}

\textcolor{red}{\uline{Warning}: If $\int_a^{\infty} f(x)\ dx$ \textit{converges}, \probonly{\underline{\hspace{3.5in}}} \solonly{we cannot say anything about the convergence of $\int_a^{\infty} g(x)\ dx$!}\\ Similarly, if $\int_a^{\infty} g(x)\ dx$ \textit{diverges}, \probonly{\underline{\hspace{3.5in}}} \solonly{we cannot say anything about the convergence of $\int_a^{\infty} f(x)\ dx$!}}
\end{framed}

\item 
\note{Try to get students to first think about what the most significant
terms in the integrand are, as a way of guessing whether the integral
converges.  Then help them with methodical ways of writing
inequalities.  It helps to emphasize basics, like: if two fractions
have the same numerator, the one with the bigger denominator is
smaller (assuming everything is positive).}

In each of the following, decide whether the improper integral
converges or diverges, and use the Comparison Theorem to justify your

\begin{enumerate}
  \item
  \begin{problem}[2.3in]
  $\displaystyle \int_1^\infty \frac{x}{\sqrt{3 + x^5}}~dx$.
  \end{problem}

    \begin{solution} % originally by Henry Shin, modified by Janet
    First, we should decide whether we \textit{think} the integral
    converges; then, we'll use the Comparison Theorem to justify.  Here
    (in \ideacolor) is a reasonable thought process.

    \ideas{When deciding whether such an improper integral converges,
    we're really looking to see what happens when $x$ gets really
    large (the ``improperness'' of the integral comes from the fact
    that $x$ is allowed to increase without bound).  In the
    denominator, when $x$ is large, $3$ is miniscule compared to
    $x^5$, so it seems that the integrand is kind of similar to
    $\frac{x}{\sqrt{x^5}} = \frac{1}{x^{3/2}}$ when $x$ is large.
    Since we know by {{\#2}(c)} that $\displaystyle
    \int_1^\infty \frac{1}{x^{3/2}}~dx$ converges, this suggests that
    $\displaystyle \int_1^\infty \frac{x}{\sqrt{3 + x^5}}~dx$
    converges as well.}

    To \textit{justify} this, we'll use the Comparison Theorem.  Since
    we are trying to show that our integral converges, we need to come
    up with a function that is \textit{larger} than $\displaystyle
    \frac{x}{\sqrt{3 + x^5}}$.  A good first guess is
    $\dfrac{1}{x^{3/2}}$, since that's what we used in our thought
    process.  But we need to make sure: is $\dfrac{1}{x^{3/2}}$ really
    larger than $\displaystyle \frac{x}{\sqrt{3 + x^5}}$?  Well,
    \begin{align*}
    \frac{x}{\sqrt{3 + x^5}} & \leq \frac{x}{\sqrt{x^5}} \textrm{
    (left side has bigger denominator, same numerator)} \\
    &= \frac{1}{x^{3/2}}
    \end{align*}
    So, here is a complete justification:      

      \fbox{%
        \parbox[t]{\linewidth-7pt}{%
        When $x \geq 1$,
        \[ 0 < \frac{x}{\sqrt{3 + x^5}} < \frac{x}{\sqrt{x^5}} =
        \frac{1}{x^{3/2}}.\footnotemark\] We know by
        {{\#2}(c)} that $\displaystyle \int_1^{\infty}
        \frac{1}{x^{3/2}}~dx$ converges.{\footnotemark} Therefore,
        $\displaystyle \int_1^{\infty} \frac{x}{\sqrt{3 + x^5}}~dx$
          \textbf{converges} by the Comparison Theorem.%
          }
          }\addtocounter{footnote}{-1}\footnotetext{When using the Comparison
          Theorem, it's very important to state the inequality you are using
          to compare.  It's also important to state that the two functions
          you're comparing are both non-negative, which we've done by
          putting the $\geq 0$ part of the inequality.  Stating these things
          explains why it's valid to apply the Comparison Theorem.}
          \stepcounter{footnote}\footnotetext{It's also very important to say
          how you know whether the integral you're comparing to converges or
          diverges.  Sometimes you can simply cite a result; other times,
          you'll have to actually evaluate the integral.}      
          \end{solution}

          \item
          \begin{problem}[2.3in]
          $\displaystyle \int_1^\infty \frac{1}{x^2 + \sqrt{x}}~dx$.
          \end{problem}

          \begin{solution}
          Again, let's first guess whether the integral in question converges.
          \ideas{When $x$ is really large, $x^2$ is a lot more significant
          than $\sqrt{x}$, so $x^2 + \sqrt{x}$ is kind of like $x^2$.
          Therefore, $\dfrac{1}{x^2 + \sqrt{x}}$ is similar to
          $\dfrac{1}{x^2}$ when $x$ is really large.  Since we know by
          {{\#1}(a)} that $\displaystyle \int_1^\infty
          \frac{1}{x^2}~dx$ converges, it seems reasonable to guess that
          $\displaystyle \int_1^\infty \frac{1}{x^2 + \sqrt{x}}~dx$
          converges as well.}  In fact, it does, and here is a complete
          justification:

      \fbox{%
        \parbox[t]{\linewidth-7pt}{%
        We know that $\displaystyle 0 \leq \frac{1}{x^2 + \sqrt{x}} \leq
        \frac{1}{x^2}$ for $x \geq 1$.  We also know by
        {{\#1}(a)} that $\displaystyle \int_1^\infty
        \frac{1}{x^2}~dx$ converges.  So, the Comparison Theorem says
        that $\displaystyle \int_1^\infty \frac{1}{x^2 + \sqrt{x}}~dx$
          \textbf{converges} as well.%
          }
          }
          \end{solution}
        \end{enumerate}

  \item  Decide whether the following improper integrals converge or diverge.
        Explain your reasoning thoroughly.
        \begin{enumerate}
          \item
          \note{Although this is easy to compute directly, it's worth talking
          about how knowing about the convergence of $\displaystyle
          \int_1^\infty \frac{1}{x}~dx$ is really all that's necessary, since
          $\displaystyle \int_1^5 \frac{1}{x}~dx$ is a definite integral and
          hence just a number that doesn't affect convergence.}

          \begin{problem}[\vfill]
          $\displaystyle \int_5^\infty \frac{1}{x}~dx$.
          \end{problem}

          \begin{solution}
          We can compute this directly using the definition:
          \begin{align*}
          \int_5^\infty \frac{1}{x}~dx &= \lim_{b \to \infty} \int_5^b
          \frac{1}{x}~dx \\
          &= \lim_{b \to \infty} \left( \ln |x| \Big|_5^b \right) \\
          &= \lim_{b \to \infty} (\ln |b| - \ln 5) \\
          &= \infty \textrm{ (diverges)}
          \end{align*}
          We could also have figured this out using what we know about
          $\displaystyle \int_1^\infty \frac{1}{x}~dx$; we showed in
          {{\#1}(b)} that this diverges.  Graphically, we can think
          of $\displaystyle \int_1^\infty \frac{1}{x}~dx$ as the area under $y
          = \frac{1}{x}$ to the right of $x = 1$; similarly, $\displaystyle
          \int_5^\infty \frac{1}{x}~dx$ represents the area under $y =
          \frac{1}{x}$ to the right of $x = 5$.  The difference between the
          two is the area under $y = \frac{1}{x}$ between $x = 1$ and $x = 5$;
          that area is clearly finite (it doesn't go on forever in any
          direction), so $\displaystyle \int_1^\infty \frac{1}{x}~dx$ and
          $\displaystyle \int_5^\infty \frac{1}{x}~dx$ must both be finite or
          both be infinite.

          If we want to write this observation mathematically, we could say
          that
          \[ \int_1^\infty \frac{1}{x}~dx = \int_1^5 \frac{1}{x}~dx +
          \int_5^\infty \frac{1}{x}~dx.\]
          The integral $\displaystyle \int_1^5 \frac{1}{x}~dx$ is a definite
          integral, so it's just a number.  Therefore, since we know that
          $\displaystyle \int_1^\infty \frac{1}{x}~dx$ diverges (it's
          $\infty$), this equation tells us that $\displaystyle \int_5^\infty
          \frac{1}{x}~dx$ must diverge as well.
          \end{solution}


          \item
          \begin{problem}[\vfill]
          $\displaystyle \int_4^\infty \frac{x^2}{x^3 - 1}~dx$.
          \end{problem}

          \begin{solution}
          We can compute this directly (integrate by substituting $u = x^3 -
          1$), or use comparison.  This solution will use comparison.

          When $x \geq 4$, \[\frac{x^2}{x^3 - 1} > \frac{x^2}{x^3} =
          \frac{1}{x} > 0.\] We know by {{\#1}(b)} that
          $\displaystyle \int_1^\infty \frac{1}{x}~dx$ diverges, and the
          reasoning in {{\#5}(a)} shows that
          $\displaystyle \int_4^\infty \frac{1}{x}~dx$ diverges as well.  By
          the Comparison Theorem, $\displaystyle \int_4^\infty \frac{x^2}{x^3
          - 1}~dx$ diverges.
          \end{solution}

          
          \item
          \begin{problem}[\vfill]
          $\displaystyle \int_1^\infty \frac{e^{-1/x}}{x^3}~dx$.
          \end{problem}

          \begin{solution}
          This integrand looks like it could be hard to integrate, so let's
          try comparison.  In the expression $e^{-1/x}$, $e$ is always being
          raised to a negative power (since we are concerned with $x \geq 1$),
          so $e^{-1/x} < 1$.  Therefore, when $x \geq 1$, 
          \[ 0 < \frac{e^{-1/x}}{x^3} < \frac{1}{x^3}.\]
          We know by {{\#2}(a)} that $\displaystyle \int_1^\infty
          \frac{1}{x^3}~dx$ converges, so the Comparison Theorem tells us that
          $\displaystyle \int_1^\infty \frac{e^{-1/x}}{x^3}~dx$ converges as
          well. 
          \end{solution}
          
          
        \end{enumerate}



        
    

            
\end{enumerate}
