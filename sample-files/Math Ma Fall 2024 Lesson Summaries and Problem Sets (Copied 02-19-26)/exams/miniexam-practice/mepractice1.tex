\item \points{14} \textbf{Algebra.} 

  \begin{enumerate}
  \item 

    \begin{problem}[\stretch{0.5}]
      Simplify $\left(\frac{1}{8}\right)^{2/3} + 5^0 + 7^{-1}$
    \end{problem}

    \begin{solution}
      $\displaystyle \left(\frac{1}{8}\right)^{2/3} = \left[\left( \frac{1}{8} \right)^{1/3} \right]^2 = \left( \frac{1}{2} \right)^2 = \frac{1}{4}$, so
      \begin{align*}
        \left(\frac{1}{8}\right)^{2/3} + 5^0 + 7^{-1} &= \frac{1}{4} + 1 + \frac{1}{7} \\
        &= \frac{7}{28} + \frac{28}{28} + \frac{4}{28} \\
        &= \boxed{\frac{39}{28}}
      \end{align*}
    \end{solution}

  \item  % Fall 2019 Practice Algebra Gateway 2

    \begin{problem}[\vfill]
      Simplify $\dfrac{\sqrt{64 x^{-4}y^{3}}}{-2xyz^{-1}}$, assuming $x, y, z$ are positive numbers. Your final expression should \textbf{NOT} involve any radical (root) signs, negative exponents, or fractions of fractions.
    \end{problem}

    \begin{solution}
      \begin{align*}
        \frac{\sqrt{64 x^{-4}y^{3}}}{-2xyz^{-1}}  &=
        \frac{ \left( 64 x^{-4}y^{3} \right)^{1/2}}{-2xyz^{-1}} \\ 
        &= \frac{64^{1/2} (x^{-4})^{1/2} (y^3)^{1/2} }{-2xyz^{-1}} \\      
        &= \frac{8 x^{-2}y^{\frac{3}{2}}}{-2xyz^{-1}}\\
        &= -4 x^{-2 - 1} y^{\frac{3}{2} - 1} z^{0 - (-1)} \\
        &= -4 x^{-3} y^{1/2} z \\
        &= \boxed{-\frac{4 y^{1/2} z}{x^3}}
      \end{align*}
    \end{solution}

  \item % Fall 2019 Practice Algebra Gateway 2

    \begin{problem}[\vfill\newpage]
      Solve the equation $\dfrac{1}{2x+1} = \dfrac{x-1}{x}$ for $x$. 
    \end{problem}

    \begin{solution}
      Let's multiply both sides by $x(2x+1)$ to get rid of the fractions: 
      \begin{align*}
        x(2x + 1) \dfrac{1}{2x+1} &= x (2x + 1) \dfrac{x-1}{x} \\
        x& =(2x+1) (x-1) \\ 
        x &= 2x^2-x-1 \\
        \intertext{This isn't linear in $x$ (if you aren't sure what this means, see \href{https://people.math.harvard.edu/~jjchen/mathma/handouts/solving-equations.html}{the ``Algebra Practice: Solving equations'' handout}), so we'll move everything to one side of the equation.}
        0 &= 2x^2 - 2x - 1 
      \end{align*}
      I don't see a way to factor this, so I'll use the quadratic
      formula: 
      \begin{align*} 
        x &= \frac{2 \pm \sqrt{4-4 (2)(-1)}}{2 \cdot 2} \\ 
        & = \frac{2 \pm \sqrt{12}}{4}\\
        & = \boxed{\frac{1 \pm \sqrt{3} }{2}}
      \end{align*}
    \end{solution}

  \item  % Fall 2019 Practice Algebra Gateway 2

    \begin{problem}[\vfill]
      Simplify $\dfrac{a + b^{-2}}{a - \displaystyle \frac{2b}{a^3}}$ so that it does \textbf{NOT} contain fractions in the numerator or denominator or involve any negative exponents.
    \end{problem}

    \begin{solution}
      We have
      \begin{align*}
        \frac{a + b^{-2}}{a - \displaystyle \frac{2b}{a^3}} &= \frac{a + \dfrac{1}{b^2}}{a - \displaystyle \frac{2b}{a^3}} \\
        \intertext{Multiplying an expression by 1 doesn't change it, so we can rewrite this as} %
        &= \frac{a + \dfrac{1}{b^2}}{a - \displaystyle \frac{2b}{a^3}} \cdot \frac{a^3 b^2}{a^3 b^2} \\
        \intertext{(Here, we've written $1$ as $\dfrac{a^3 b^2}{a^3 b^2}$ so that we can get rid of the fractions in the numerator and denominator.)} %
        &= \boxed{\frac{a^4 b^2 + a^3}{a^4 b^2 - 2b^3}}
      \end{align*}
    \end{solution}
  \end{enumerate}

\item \points{3} 

  \begin{problem}[\vfill]
    Sketch the graph of $f(x) = \dfrac{x^3 + 3x^2 - 4x}{x (x - 1)}$. Explain briefly how you got it. 
  \end{problem}

  \solonly{\related{{{\#4}} on \href{https://people.math.harvard.edu/~jjchen/mathma/docs/worksheets/worksheet02-sols.html}{the worksheet ``A Library of Basic Functions and Properties''}; \href{https://people.math.harvard.edu/~jjchen/mathma/docs/homework/PS02.html}{Problem Set 2}, {{\#5}}}}

  \begin{solution}
    The domain of $f$ is $x \neq 0, 1$. Let's factor the numerator:
    \[ f(x) = \dfrac{x (x^2 + 3x - 4)}{x(x - 1)} = \dfrac{x (x - 1)(x + 4)}{x (x - 1)}.\]
    When $x \neq 0, 1$, we can cancel $x(x-1)$ in the numerator and denominator to rewrite $f(x) = x + 4$. So, $f$ is undefined at $x = 0, 1$ but otherwise looks just like the graph of $x + 4$. Thus, the graph of $f$ looks like this:
    \begin{center}
      \answer{
        \begin{tikzpicture}
          \begin{axis}[xtick={-4, 0, 1}, ytick={4}, axis equal image, width=2.5in]
            \addplot+[domain=-6:6] {x + 4};
            \addplot+[open] coordinates {(0, 4) (1, 5)}; 
          \end{axis}
        \end{tikzpicture}
      }
    \end{center}
  \end{solution}

  \begin{enumerate}
  \item

    \begin{problem}[\vfill]
      Sketch the graph of a function $f$ with domain $[0, \infty)$ that has \uline{all} of the following properties:
      \begin{itemize}
      \item $f$ is continuous on its entire domain
      \item $f(0) = 3$
      \item $f$ is increasing on $(0, 8)$ and decreasing on $(8, \infty)$
      \item $f$ is concave up on $(0, 5)$ and concave down on $(5, \infty)$
      \end{itemize}
    \end{problem}

    \solonly{\related{\href{https://people.math.harvard.edu/~jjchen/mathma/docs/homework/PS02.html}{Problem Set 2}, {{\#3}}}}

    \begin{solution}
      On $(0, 5)$, the function should be increasing and concave up; on $(5, 8)$, it should be increasing and concave down; on $(8, \infty)$, it should be decreasing and concave down. Here's one possibility; we've colored the three pieces differently just to make the features of each more clear.
      \begin{center}
        \answer{
          \begin{tikzpicture}
            \begin{axis}[xtick={5, 8}, ytick={3}]
              \addplot+[domain=0:5, red] {x^2 / 10 + 3}; \addplot+[domain=5:8, green!70!black] {-(x-8)^2/6+7}; \addplot+[domain=8:15, blue] {-(x-8)^2/6+7};
            \end{axis}
          \end{tikzpicture}
        }
      \end{center}
    \end{solution}

  \item
    \begin{problem}[\stretch{0.5}]
      How many $x$-intercepts could a function satisfying the properties in (a) have? List all of the possibilities, and explain briefly.
    \end{problem}

    \begin{solution}
      Such a function must have exactly \fbox{one} $x$-intercept. Since $f(0) = 3$ and the function increases on $(0, 8)$, it can't have any $x$-intercepts on $[0, 8]$. After that, the function decreases, and since it's concave down, it must cross the $x$-axis at some point. 
    \end{solution}

  \end{enumerate}

  % Juliana (Goal Topic: Average Rate of Change/Secant Lines) 

  In Houston, Texas, monthly water bills increased from January 2010 to January 2012, but the rate of increase slowed during this time period. 
  In January 2010, the water bill for a typical medium-use family (a
  family of four using 100 gallons per person per day) was \$$39$.  In
  January 2012, the water bill for the same family was \$$53$.
  \begin{enumerate}
  \item
    \begin{problem}[\vfill]
      Compute the average rate of change of the water bill for a typical
      medium-use family from January 2010 to January 2012.  Please specify
      the units of your answer.
    \end{problem}

    \begin{solution}
      The average rate of change is $\dfrac{\text{change in water
          bill}}{\text{change in time}} = \dfrac{\$53 - \$39}{2012 - 2010}
      = \dfrac{\$14}{2\ \text{years}} = \boxed{\$7\ \text{per year}}$.
    \end{solution}
  \end{enumerate}

  Let $W(t)$ be the water bill (in dollars) for a typical medium-use family in Houston $t$ years after January 2010.

  \begin{enumerate}[resume]
  \item
    \begin{problem}[\stretch{3}]
      Sketch a possible graph of $W(t)$, labeling any points whose coordinates you know. 
    \end{problem}

    \solonly{\related{\#3, {{\#4}} on \href{https://people.math.harvard.edu/~jjchen/mathma/docs/worksheets/worksheet01-sols.html}{the worksheet ``Introduction to Functions''}; \href{https://people.math.harvard.edu/~jjchen/mathma/docs/homework/PS01.html}{Problem Set 1}, {{\#2}}}}

    \begin{solution} 
      The key piece of information we were given, besides the typical
      water bills in January 2010 and January 2012, was that water bills were increasing, but the rate of increase slowed during this time. That means, for example, that the increase in each month is less than the increase in the month before, so here's a possible graph.
      \begin{center}
        \answer{
          \begin{tikzpicture}
            \begin{axis}[xtick={0, 1, 2}, ytick={39, 53}, ymin=0, height=2in, width=2in, enlarge x limits=auto, clip=false]
              \addplot+[domain=0:2]{-2*x^2+11*x+39};
              \addplot+[closed] coordinates {(0, 39)} node[anchor=north west]{$(0, 39)$};
              \addplot+[closed] coordinates {(2, 53)} node[right]{$(2, 53)$};
            \end{axis}
          \end{tikzpicture}
        }
      \end{center}
    \end{solution}

  \item

    \begin{problem}[\nstretch{3}]

      Arrange the following four quantities in increasing order:

      \begin{enumeratecols}[label=\Alph*.]{4}
      \item zero $\vphantom{\dfrac{1}{1}}$
      \item $W(1) - W(0)$ $\vphantom{\dfrac{1}{1}}$
      \item $\dfrac{W(2)-W(1)}{2-1}$
      \item $\dfrac{W(2)-W(0)}{2}$
      \end{enumeratecols}

      (For example, your answer could be $\text{D} < \text{C} < \text{B} < \text{A}$.) Justify your answer.
    \end{problem}

    \begin{solution}
      We can interpret all of these quantities as slopes of secant lines on the graph of $W(t)$. (It may help to rewrite $W(1) - W(0) = \dfrac{W(1) - W(0)}{1 - 0}$.) 
      \begin{center}
        \begin{tikzpicture}
          \begin{axis}[xtick={0, 1, 2}, ytick={39, 53}, ymin=0, enlarge x limits=auto, width=3.5in, height=3.5in, cycle list name=mycolor]
            \addplot+[domain=0:2]{-2*x^2+11*x+39}; %
            \addplot+[domain=0:1] {9*x+39} node[pos=0.5, anchor=south east]{B}; %
            \addplot+[domain=1:2] {5*(x-1)+48} node[pos=0.5, above]{C}; %
            \addplot+[domain=0:2] {7*x+39} node[pos=0.5, below] {D}; %
          \end{axis}
        \end{tikzpicture}
      \end{center}
      Comparing the two lines passing through $(0, 39)$, we see that D $<$ B. Comparing the two lines passing through $(2, 53)$, we see that C $<$ D. Putting these together, C $<$ D $<$ B. Finally, all of the slopes pictured are positive, so Choice A is the smallest: \fbox{ A $<$ C $<$ D $<$ B}.
    \end{solution}    

  \end{enumerate}

  \begin{enumerate}

    \begin{problem}
      Sketch the graph of $f(x) = \sqrt{x - 1} - 2$. Make sure to label the axes in a way that makes the domain and range clear. 
    \end{problem}

    \probonly{
      \axes[width=4.5in]
    }

    \begin{solution}
      We start with $y = \sqrt{x}$, shift it right by 1, and shift it down by 2: 
      \begin{center}
        \answer{
          \begin{tikzpicture}
            \begin{axis}[xtick={1}, ytick={-2}, xmin=0, axis equal image, width=3in]
              \addplot+[domain=1:1.5] {sqrt(x - 1) - 2};%
              \addplot+[domain=1.5:7] {sqrt(x - 1) - 2};%
              \addplot+[closed] coordinates {(1, -2)};% 
            \end{axis}
          \end{tikzpicture}
        }
      \end{center}
    \end{solution}

  \item
    \begin{problem}
      On the axes in (a), sketch the graph of $g(x) = 3 f(x)$ using a dashed line so that it's easy to tell which graph is $f$ and which is $g$. Again, make sure to label the axes in a way that makes the domain and range clear.
    \end{problem}

    \begin{solution}
      To get the graph of $g$, we stretch the graph of $f$ vertically by a factor of 3: 
      \begin{center}
        \answer{
          \begin{tikzpicture}
            \begin{axis}[xtick={1}, ytick={-6, -2}, xmin=0, axis equal image, width=3in]
              \addplot+[domain=1:1.5] {sqrt(x - 1) - 2};%
              \addplot+[domain=1.5:7] {sqrt(x - 1) - 2};%
              \addplot+[closed] coordinates {(1, -2)}; %
              \addplot+[domain=1:7, dashed] {3 * (sqrt(x - 1) - 2)}; %
              \addplot+[closed] coordinates {(1, -6)}; % 
            \end{axis}
          \end{tikzpicture}
        }
      \end{center}
    \end{solution}

  \item
    \begin{problem}[\vfill]
      Find the coordinates of any points of intersection. 
    \end{problem}

    \begin{solution}
      We can see that the graphs of $f$ and $g$ intersect at the $x$-intercept of $f$, so let's find that:
      \begin{align*}
        \sqrt{x - 1} - 2 &= 0 \\
        \sqrt{x - 1} &= 2 \\
        \shortintertext{Squaring both sides,}
        x - 1 &= 4 \\
        x &= 5 
      \end{align*}
      So, the intersection point is $\boxed{(5, 0)}$.

      \emph{Note:} When we square both sides of an equation, we can sometimes get extraneous solutions, so we should generally plug any solutions we plug in back to the original equation to make sure they work. Here, though, we know from our graph that there should be exactly one intersection, so the one solution we found must truly be a solution. 
    \end{solution}
  \end{enumerate}
