\item Let $f(x) = x^3$. In this problem, we'll look at $f'(1)$.

  \begin{enumerate}
  \item

    \begin{grading}
      2 points: 1 for the graph of $y = x^3$, 1 for sketching an appropriate tangent line 
    \end{grading}

    \begin{problem}[\stretch{1.25}]
      Sketch $y = x^3$ together with a line whose slope is $f'(1)$.
    \end{problem}

    \begin{solution}
      $f'(1)$ is the slope of the line tangent to the graph at $x=1$. 
      
      \begin{tikzpicture}
        \begin{axis}
          \addplot[domain=-1.5:1.5]{x^3};
          \addplot[red, domain=-0.5:1.5]{3*(x-1)+1};
        \end{axis}
      \end{tikzpicture}
    \end{solution}

  \item \label{approximate-derivative-values-of-h}

    \begin{grading}
      1 point: 0.25 for saying 0, 0.75 for explanation 
    \end{grading}

    \begin{problem}[0.75in]
      In class, we saw that we can \uline{approximate} $f'(1)$ using slopes of secant lines. If we want to approximate $f'(1)$ using the slope of the secant line between $(1, f(1))$ and $(1 + h, f(1 + h))$, we expect to get a more accurate approximation if $h$ is close to what number? Explain briefly.
    \end{problem}

    \begin{solution}
      To approximate the slope of a tangent line using the slope of a secant
      line, we want the two points where the secant line intersects the graph
      to be close together. Since $h$ is the change in $x$ between the two
      points, we want $h$ to be close to zero. 
    \end{solution}

  \calcitem \label{approximate-derivative-estimates}

    \begin{grading}
      2 points: 1 for each approximation. You don't need to check for accuracy, since they will all have different values of $h$; just check that the results are close to 3. 
    \end{grading}

    \begin{problem}[\vfill\newpage]
      Pick two values of $h$ that are very close to your answer to \ref{approximate-derivative-values-of-h}, one that's greater than your answer to \ref{approximate-derivative-values-of-h} and one that's less than your answer to \ref{approximate-derivative-values-of-h}. For each one, find the slope of the secant line between $(1, f(1))$ and $(1 + h, f(1 + h))$.
    \end{problem}

    \begin{solution}
      We want $h$ to be close to zero, so we can take $h=0.1$ and $h=-0.1$.
      First, if $h=0.1$, then $f(1+0.1)=1.1^3=1.331$. So the slope
      of the secant line between $(1,1)$ and $(1.1, 1.331)$ is
      \[
        \frac{1.331-1}{1.1-1}=\frac{0.331}{0.1}=3.31.
      \]
      If $h=-0.1$, then $f(1-0.1)=0.9^3=0.729$. So the slope of the secant
      line between $(1, 1)$ and $(0.9, 0.729)$ is
      \[
        \frac{1-0.729}{1-0.9}=\frac{0.271}{0.1}=2.71.
      \]
    \end{solution}

  \item

    \begin{grading}
      3 points:
      \begin{itemize}[nosep]
      \item 2 for simplifying $\dfrac{f(1 + h) - f(1)}{h}$
      \item 0.5 for for using limit notation correctly (writing $\displaystyle \lim_{h \to 0}$ on each line until they actually take the limit)
      \item 0.5 for correct limit 
      \end{itemize}

    \end{grading}

    \begin{problem}[\vfill]
      Calculate $f'(1)$ using the definition of the derivative. (\check Make sure your answer is consistent with your answers to \ref{approximate-derivative-estimates}!)
    \end{problem}

    \begin{solution}
      First, we can write down a $f'(1)$ as a 
      limit using the definition of the derivative.
      \begin{align*}
        f'(1) &= \lim_{h \to 0} \frac{f(1+h)-f(1)}{h} \\
        &= \lim_{h \to 0} \frac{(1+h)^3 - 1}{h} \\
        \intertext{Now we expand $(1+h)^3$ and combine like terms.}
        &= \lim_{h \to 0} \frac{1 + 3h + 3h^2 + h^3 - 1}{h} \\
        &= \lim_{h \to 0} \frac{3h + 3h^2 + h^3}{h} \\
        \intertext{Now we factor and cancel $h$ in the numerator and
        denominator.}
        &= \lim_{h \to 0} \frac{\cancel{h}(3+3h+h^2)}{\cancel{h}} \\
        &= \lim_{h \to 0} (3 + 3h + h^2) \\
        \intertext{We can now determine the value that $3+3h+h^2$ approaches
        as $h$ approaches zero.}
        &= 3+3(0)+0^2 = \boxed{3}
      \end{align*}
    \end{solution}

  \item

    \begin{grading}
      1 point
    \end{grading}

    \begin{problem}[\nstretch{0.3}]
      Find the equation of the line tangent to $y = x^3$ at $x = 1$. 
    \end{problem}

    \begin{solution}
      We know that the slope of the line is $f'(1)=3$, and it passes
      through the point $(1, 1)$. So the equation of the line
      in point-slope form is \fbox{$y-1=3(x-1)$.}
    \end{solution}

\item  Let $f(x) = \dfrac{1}{x^2} $. 

  \begin{enumerate} 

  \item

    \begin{grading} 
      4 points:
      \begin{itemize}[nosep]
      \item 3 for correctly simplifying $\dfrac{ f(-2+h) - f(-2) }{h}$
      \item 0.5 for using limit notation correctly (writing $\displaystyle \lim_{h \to 0}$ on each line until they actually take the limit)
      \item 0.5 for correct limit 
      \end{itemize}
    \end{grading} 

    \begin{problem}[\nstretch{1.5}]
      Use the limit definition of the derivative to calculate $f'(-2)$.
    \end{problem}

    \begin{solution}
      First, we can use the definition of the derivative to write $f'(-2)$
      as a limit.
      \begin{align*} 
        f'(-2) &=\lim_{ h \to 0 } \frac{ f(-2+h) - f(-2) }{(-2+h )- (- 2)} \\
        &= \lim_{h \to 0} \frac{ \dfrac{1}{(-2+h)^2} - \dfrac{1}{4}}{h}\\
        \intertext{Now we simplify the complex fraction.}
        &= \lim_{h \to 0} \frac{ \dfrac{1}{(-2+h)^2} - \dfrac{1}{4}}{h}
        \cdot \frac{4(-2+h)^2}{4(-2+h)^2} \\
        &=
        \lim_{ h \to 0 } \frac{ 4-(-2+h)^2}{4h(-2+h)^2}\\
        \intertext{Now we can expand $(-2+h)^2$ and combine like terms.}
        & =
        \lim_{ h \to 0 } \frac{ 4-(4-4h+h^2)}{4h\cdot(-2+h)^2} \\
        & =
        \lim_{ h \to 0 } \frac{ 4h-h^2}{4h(-2+h)^2}\\
        \intertext{Now we factor and cancel $h$ in the numerator and
        denominator.}
        & =
        \lim_{ h \to 0 } \frac{\cancel{h}(4-h)}{4\cancel{h}(-2+h)^2}\\
        & =
        \lim_{ h \to 0 } \frac{ 4-h}{4(-2+h)^2} \\
        \intertext{We can now determine the value that $\dfrac{4-h}{4(-2+h)}$
        approaches as $h$ approaches zero.}
        & = \frac{4-0}{4(-2+0)^2} = \frac{4}{16} = \boxed{\frac{1}{4}}
      \end{align*} 
    \end{solution} 

  \item

    \begin{grading}
      2 points: 1 for graph, 1 for some sort of consistency check
    \end{grading}

    \begin{problem}[\vfill]
      \check Sketch the graph of $f(x) = \dfrac{1}{x^2}$, and explain
      in a sentence how your calculation of $f'(-2)$ agrees with the
      graph. 
    \end{problem}

    \begin{solution}
      Here's the graph. Since $f'(-2)$ represents the slope of the tangent line at $x = -2$, we can see that $f'(-2)$ should be positive, which agrees with our calculation.
      \begin{center}
        \begin{tikzpicture}
          \begin{axis}[ymax=6, xtick=\empty, ytick=\empty]
            \addplot+[domain=-3:3] {1/x^2};
          \end{axis}
        \end{tikzpicture}
      \end{center}
    \end{solution}

  \item

    \begin{grading}
      2 points: 1 for the answer, 1 for the explanation. They could,
      of course, also use the definition of derivative to calculate
      $f'(2)$. 
    \end{grading}

    \begin{problem}[\nstretch{1.5}]
      Find $f'(2)$, and explain how you got it. 
    \end{problem}

    \begin{solution}
      Since the graph exhibits even symmetry, 
      we can see that $f'(2)$ (the slope of the tangent line
      at $x = 2$) should be the opposite of $f'(-2)$, so $f'(2) =
      \boxed{- \frac{1}{4}}$. 
    \end{solution}

  \end{enumerate}

      \item The graph is concave down because she runs more and more slowly throughout the hour.
      \end{itemize}

      \end{center}
    \end{solution}

  \item \label{gottlieb-4-4-11b}

    \begin{workshop}

    \begin{problem}[\nstretch{2.5}]
      Suppose that at 8:00 am, the runner is running at a speed of 9 miles per hour. Find good upper and lower bounds for the total distance she has run by 8:30 am. Explain your reasoning with both words and a graph. (\emph{Hint:} Think about what you learned in this class and the previous class!)
    \end{problem}

    \begin{grading} 
      6 points: 
      \begin{itemize}[nosep]
      \item 2 for a correct secant line approximation

      \item 2 for a correct tangent line approximation
      \item 1 for explaining why the tangent line approximation is an upper bound 
      \end{itemize} 
    \end{grading} 

    \begin{solution} 
      We know two points on the graph, so one thing we can do is a secant line approximation. The secant line through the points $(8, 10)$ and $(9, 16)$ has slope 6, so its equation is $d - 10 = 6 (t - 8)$, or $d = 6 (t - 8) + 10$. So, the point on this line with $t = 8.5$ has $d = 6 (0.5) + 10 = 13$. Here's a picture illustrating this approximation:
      \begin{center}
        \begin{tikzpicture}
          \begin{axis}[xtick={8, 9}, axis x discontinuity=parallel, ytick={10, 16}, xmin=7.2, ymin=0, xlabel={$t$ (hours since midnight)}, ylabel={$d$ (miles)}, clip=false, enlarge x limits=auto]
            \addplot+[domain=8:9] {4 + 6*sqrt(3)*sqrt(-23/3 + x)};
            \addplot+[closed] coordinates { (8, 10) } node[below]{$(8, 10)$};
            \addplot+[closed] coordinates { (9, 16) } node[right] {$(9, 16)$ };
            \addplot+[sharp plot, skyblue] coordinates { (8, 10) (9, 16)};
            \addplot+[closed, skyblue] coordinates { (8.5, 13) } node[anchor=north west] {$(8.5, 13)$}; 
          \end{axis}
        \end{tikzpicture}
      \end{center}
      We can see from the graph that this is a lower bound for the actual value, since the secant line is under the graph. 

      We also know that, at 8:00 am, the runner is running at 9 mph. That means the tangent line at $t = 8$ has slope 9, so its equation is $d - 10 = 9 (t - 8)$, or $d = 9 (t - 8) + 10$. The point on this line with $t = 8.5$ has $d = 9(0.5) + 10 = 14.5$. Here's a picture illustrating this approximation:
      \begin{center}
        \begin{tikzpicture}
          \begin{axis}[xtick={8, 9}, axis x discontinuity=parallel, ytick={10, 16}, xmin=7.2, ymin=0, xlabel={$t$ (hours since midnight)}, ylabel={$d$ (miles)}, clip=false, enlarge x limits=auto]
            \addplot+[domain=8:9] {4 + 6*sqrt(3)*sqrt(-23/3 + x)};
            \addplot+[closed] coordinates { (8, 10) } node[below]{$(8, 10)$};
            \addplot+[closed] coordinates { (9, 16) } node[right] {$(9, 16)$ };
            \addplot+[sharp plot, hotpink] coordinates { (8, 10) (8.6, 15.4)};
            \addplot+[closed, hotpink] coordinates { (8.5, 14.5) } node[anchor=south east] {$(8.5, 14.5)$}; 
          \end{axis}
        \end{tikzpicture}
      \end{center}
      We can see from the graph that 14.5 is an upper bound for the actual value, since the tangent line is above the graph. 

      So, \fbox{13 is a lower bound, and 14.5 is an upper bound}. In other words, the distance she has run by 8:30 am is between $13$ and $14.5$.
    \end{solution}
  \end{enumerate}

\item \label{fall2018-premidterm-practice3-3-modified} 

  A company is designing a box with no top to transport fruit. The specifications it must meet are:
  \begin{itemize}[nosep]
  \item The box must have a volume of 3 cubic feet.
  \item The bottom of the box is a rectangle where one side is twice the length of the other side.
  \item The bottom of the box is made of material that costs 70 cents per square foot, while the sides are made of material that costs 20 cents per square foot.
  \end{itemize}

  \begin{enumerate}
  \item

    \begin{grading}
      2 points: 0.5 for finding the height, 1 for finding the cost of the sides, 0.5 for findign the cost of the bottom
    \end{grading}

    \begin{problem}[\vfill]
      If they decide to make the bottom of the box 1 ft $\times$ 2 ft, how much will the box cost? 
    \end{problem}

    \begin{solution}
      Let's draw a picture.
      
      \begin{tikzpicture}
        \pgfmathsetmacro{\x}{3}
        \pgfmathsetmacro{\y}{2}
        \pgfmathsetmacro{\z}{1.5}
        \path (0,0,\y) coordinate (A) (\x,0,\y) coordinate (B) (\x,0,0) coordinate (C) (0,0,0)
          coordinate (D) (0,\z,\y) coordinate (E) (\x,\z,\y) coordinate (F) (\x,\z,0) coordinate (G)
          (0,\z,0) coordinate (H);
        \draw (A)--(B)--(C)--(G)--(F)--(B) (A)--(E)--(F)--(G)--(H)--(E);
        \draw (A)--(D)--(C) (D)--(H);

        \draw[thin,|<->|] ($(A)+(0,-4pt)$) -- node[below]{2 feet}($(B)+(0,-4pt)$);
        \draw[thin,|<->|] ($(B)+(-45:4pt)$) -- node[below,sloped]{1 foot}($(C)+(-45:4pt)$);
        \draw[thin,|<->|] ($(C)+(4pt,0)$) -- node[midway, right]{?}($(G)+(4pt,0)$);
      \end{tikzpicture}

      Since the volume of the box must be 3 cubic feet, the height of the box would need to be 1.5 feet.
      This means that the cost of the bottom of the box would be $(70\text{ cents/ft}^2)(2\text{ ft})
      (1\text{ ft})=\$1.40$, and the cost of the sides of the box would be $(20\text{ cents/ft}^2)\bigl[
      2(2\text{ ft})(1.5 \text{ft}) + 2(1\text{ ft})(1.5 \text{ ft})\bigr]=\$1.80$. The total cost of the
      box would be $\$3.20$, or 320 cents.
    \end{solution}  

  \item \label{fall2018-premidterm-practice3-3-modified-function}

    \begin{grading}
      5 points:
      \begin{itemize}[nosep]
      \item 1 for drawing and labeling a picture
      \item 1 for stating the goal
      \item 1 for writing a correct formula for cost (not yet in terms of 1 variable)
      \item 1 for using the volume equation to solve for the height in terms of the length of the shorter side
      \item 1 for getting cost in terms of the desired variable
      \end{itemize}
    \end{grading}

    \begin{problem}[\nstretch{2}]
      Express the cost (in cents) of such a box as a function of the length (in feet) of the shorter side of the bottom. (Remember the guidelines for presenting your work in such problems. If you've forgotten, look back at \href{https://people.math.harvard.edu/~jjchen/mathma/docs/homework/PS06.html}{Problem Set 6}.) 
    \end{problem}

    \begin{solution}
      The \hl{\textbf{goal}} is to express the cost of the box as a function of the length of the shorter side on the bottom.
      We \hl{\textbf{know}}:
      \begin{itemize}[nosep]
        \item The box must have a volume of 3 cubic feet.
        \item The bottom of the box is a rectangle where one side is twice the length of the other side.
        \item The bottom of the box is made of material that costs 70 cents per square foot, while the sides are made of
        material that costs 20 cents per square foot.
      \end{itemize}

      Let's \hl{\textbf{draw a picture and define variables.}}
      
      \begin{tikzpicture}
        \pgfmathsetmacro{\x}{3}
        \pgfmathsetmacro{\y}{2}
        \pgfmathsetmacro{\z}{1.5}
        \path (0,0,\y) coordinate (A) (\x,0,\y) coordinate (B) (\x,0,0) coordinate (C) (0,0,0)
          coordinate (D) (0,\z,\y) coordinate (E) (\x,\z,\y) coordinate (F) (\x,\z,0) coordinate (G)
          (0,\z,0) coordinate (H);
        \draw (A)--(B)--(C)--(G)--(F)--(B) (A)--(E)--(F)--(G)--(H)--(E);
        \draw (A)--(D)--(C) (D)--(H);

        \draw[thin,|<->|] ($(A)+(0,-4pt)$) -- node[below]{$\ell$}($(B)+(0,-4pt)$);
        \draw[thin,|<->|] ($(B)+(-45:4pt)$) -- node[below,sloped]{$w$}($(C)+(-45:4pt)$);
        \draw[thin,|<->|] ($(C)+(4pt,0)$) -- node[midway, right]{$h$}($(G)+(4pt,0)$);
      \end{tikzpicture}

      We'll use $\ell$ and $w$ to represent the two side lengths on the bottom of the box, and we'll have $w$ be the shorter
      length. We'll use $h$ to represent the height of the box.

      Now, let's \hl{\textbf{write a formula}} for the cost of the box. Since we know the cost per area of the bottom and the
      sides, we can say
      \begin{align*}
      C_\text{total} &= (70\text{ cents/ft}^2)(\text{area of bottom}) + (20\text{ cents/ft}^2)(\text{area of sides}) \\
      &= 70\ell w + 20(2\ell h + 2 wh) \\
      &= 70\ell w + 40\ell h + 40 wh
      \end{align*}
      where $C_\text{total}$ is the total cost of the box.

      To write $C_\text{total}$ as a function of $w$, we need to rewrite $\ell$ in terms of $w$ and $h$ in terms of $w$. To
      do this, we can use what we know. Since one side on the bottom is twice as long as the other, $\ell = 2w$. Since the
      volume of the box must be 3 cubic feet, $V = \ell w h = (2w)wh = 3$, so $h = \frac{3}{2w^2}$. Substituting these
      expressions for $\ell$ and $h$ in our formula, we get
       \begin{align*}
      C_\text{total} &= 70(2w)w + 40(2w)(\tfrac{3}{2w^2}) + 40 w(\tfrac{3}{2w^2}) \\
      &= 140w^2 + \frac{120}{w} + \frac{60}{w} \\
      &= \boxed{140w^2 + \frac{180}{w}.}
      \end{align*}
    \end{solution}

  \item

    \begin{grading}
      1 point
    \end{grading}

    \begin{problem}[1in]
      \check What kind of consistency check can you do? Do it. 
    \end{problem}

    \begin{solution}
      We can check if our formula matches the calculation we did in part (a). When $w=1$, 
      $C_\text{total} = 140(1^2) + \frac{180}{1} = 320$, which agrees with the cost we found in part (a). 
    \end{solution}

  \item

    \begin{grading}
      1.5 points: 0.5 for the minimum length, 1 for the maximum length
    \end{grading}

    \begin{problem}[\stretch{0.5}]
      Each side of the box must be at least 0.5 feet in order for the fruit to fit inside. Based on this, what's the domain of the function you wrote in \ref{fall2018-premidterm-practice3-3-modified-function}? Explain. 
    \end{problem}

    \begin{solution}
      If each side of the box must be at least $0.5$ feet, then we can write three inequalities: 
      $w \geq 0.5$, $\ell \geq 0.5$,
      and $h \geq 0.5$. The domain is the set of all possible values of $w$, so we should rewrite these inequalities to all
      be in terms of $w$: $w \geq 0.5$, $2w \geq 0.5$, and $\frac{3}{2w^2} \geq 0.5$. We want to isolate $w$ in the last
      two inequalities:
      \begin{align*}
          2w &\geq 0.5 & \frac{3}{2w^2} &\geq 0.5\\
          w &\geq 0.25 &  3 &\geq w^2 \\
          && \sqrt{3} &\geq w
      \end{align*}
      Taking all three inequalities into consideration, we can say that the minimum possible value of $w$ is 0.5 feet, and
      the largest possible value of $w$ is $\sqrt{3}$ feet. So the domain is \fbox{$0.5 \leq w \leq \sqrt{3}$.}
    \end{solution}

  \end{enumerate}

\end{enumerate}
