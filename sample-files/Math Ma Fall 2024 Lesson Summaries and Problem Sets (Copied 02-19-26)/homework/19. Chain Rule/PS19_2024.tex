\begin{enumerate}

  \doedfinity{19} 

\item 

  

  Differentiate each of the following functions with respect to $x$. It may be useful to
    first simplify or rewrite the given function.  (Not all of them require the Chain Rule, though you may still decide to use it!)

      \begin{enumerate}
      \item
        \begin{problem}[\vfill]
          $f(x)=\dfrac{2}{x+2}$
        \end{problem}

      \begin{solution}
        We begin by rewriting $f(x) = 2 (x+2)^{-1}$. We apply the Chain Rule to find:
        $$f'(x) = -2(x+2)^{-2}\cdot(1) = -2(x+2)^{-2}$$
      \end{solution}

    \item
      \begin{problem}[\vfill]
        $g(x)=\dfrac{2}{(x+2)^2}$
      \end{problem}

      \begin{solution}
        We begin by rewriting $g(x) = 2(x+2)^{-2}$. We apply the Chain Rule to find:
        $$g'(x) = 2\cdot (-2)(x+2)^{-3}\cdot (1) = -4(x+2)^{-3}$$
      \end{solution}

    \item
      \begin{problem}[\vfill]
        $h(x)=\sqrt{x+3}$
      \end{problem}

      \begin{solution}
        We begin by rewriting $h(x) = (x+3)^{1/2}$. We apply the Chain Rule to find:
        $$h'(x) = \frac{1}{2}(x+3)^{-1/2}$$
      \end{solution}

    \item
      \begin{problem}[\vfill\newpage]
       $j(x)=\sqrt{x^2+5x}$
      \end{problem}

      \begin{solution}
        We begin by rewriting $j(x) = (x^2+5x)^{1/2}$. We apply the Chain Rule to find:
        $$j'(x)=\frac{1}{2}\cdot (x^2+5x)^{-1/2}\cdot (2x+5)$$
      \end{solution}

    \item % 16.1.6

    \begin{problem}[\vfill]
     $k(x)=\sqrt{(x^2+5x)(x+4)^2}$

    \end{problem}

    \begin{solution}
      We begin by rewriting $k(x)=((x^2+5x)(x+4)^2)^{1/2}$. We apply the Chain Rule to find the derivative. Because of the complexity of what's under the radical, we'll need both the product rule and the Chain Rule to compute the derivative of those functions:
      $$k'(x)=\frac{1}{2}((x^2+5x)(x+4)^2)^{-1/2}\cdot[(2x+5)\cdot (x+4)^2+(x^2+5x)(2(x+4))]$$
    \end{solution}

    

    \end{enumerate}

      \item
   In each part, find $h'(x)$.  (Your answers should be in terms of $f$
  and $f'$.)

    \begin{enumerate}[topsep=0pt]
    \item
      \begin{problem}[\vfill]
        $h(x) = x^3 f(x)$
      \end{problem}

      \begin{solution}
        $\displaystyle \frac{d}{dx}
        \left(\fcolorbox{green}{white}{$\displaystyle x^ { 3} $}\cdot
          \fcolorbox{green}{white}{$\displaystyle f(x)$}\right)
        = {} x^ { 3} \cdot f'(x)+ 3 x^ { 2} f(x)$.
      \end{solution}

    \item
      \begin{problem}[\vfill]
        $h(x) = f(x)^3$
      \end{problem}

      \begin{solution}
        $\displaystyle \frac{d}{dx}
        \left(\fcolorbox{red}{white}{$\displaystyle
            \fcolorbox{blue}{white}{$\displaystyle f(x)$}^ { 3} $}\right)
        = 3 f(x)^ { 2} \cdot f'(x)$.
      \end{solution}

    \item
      \begin{problem}[\nstretch{1}]
        $h(x) = [x + f(x)]^3$
      \end{problem}

      \begin{solution}
        \begin{align*}
          \frac{d}{dx} \left(\fcolorbox{red}{white}{$\displaystyle
              \fcolorbox{blue}{white}{$\displaystyle [x+ f(x)]$}^ { 3}
              $}\right) = {} & 3 \left[x+ f(x)\right]^ { 2} \cdot
          \frac{d}{dx}\left(x+ f(x)\right) \\ 
          = {} & 3 \left[x+ f(x)\right]^ { 2} \cdot \left(1+
            f'(x)\right). 
        \end{align*}
      \end{solution}

    \item 
      \begin{problem}[\stretch{0.6}]
        $h(x) = f(f(x)^2)$
      \end{problem}

      \begin{solution}
        \begin{align*}
          \displaystyle \frac{d}{dx}
          \left(\fcolorbox{red}{white}{$\displaystyle f \left(
                \fcolorbox{blue}{white}{$\displaystyle f(x)^2$} \right)
              $}\right) &= f'(f(x)^2)\cdot \frac{d}{dx}
          \left(\fcolorbox{red}{white}{$\displaystyle
              \fcolorbox{blue}{white}{$\displaystyle f(x)$}^ { 2} $}\right)\\
          &= f'(f(x)^2)\cdot 2 f(x)\cdot f'(x).
        \end{align*}
      \end{solution}

    \end{enumerate}
    

  \item
    

  
Now that we know the Chain Rule, we actually don't need the Quotient
  Rule any more!  In this problem, you'll see why.

  \begin{enumerate}
  \item
    \begin{problem}[\vfill]
      Consider $f(x) = \displaystyle \frac{x^2 + 3}{x^3 + 4}$.  We can
      rewrite $f(x)$ as a product $f(x) = \displaystyle (x^2 + 3) \left[
        (x^3 + 4)^{-1} \right]$ and use the Product Rule and Chain Rule
      to differentiate this.  Do this.
    \end{problem}

    \begin{solution}
      \begin{align*}
        \frac{d}{dx} \left(\fcolorbox{green}{white}{$\displaystyle
            \left(x^ { 2} + 3\right)$}\cdot
          \fcolorbox{green}{white}{$\displaystyle  \left(x^3 +
              4\right)^ { -1} $}\right) = {} & \left(x^ { 2} +
          3\right)\cdot \frac{d}{dx}
        \left(\fcolorbox{red}{white}{$\displaystyle
            \fcolorbox{blue}{white}{$\displaystyle \left(x^3 + 4\right)$}^ { -1}
            $}\right)+ 2 x \left(x^3 + 4\right)^ {
          -1}  \\ 
        = {} & \left(x^ { 2} + 3\right)\cdot \left(-\left(x^3+4\right)^ { -2} \cdot \frac{d}{dx}\left(x^3 +
            4\right)\right)+ 2 x \left(x^3 +
          4\right)^ { -1}  \\ 
        = {} & \left(x^ { 2} + 3\right)\cdot \left(-\left(x^3 +
            4\right)^ { -2} \cdot 3x^2 \right)+
        2 x \left(x^3 + 4\right)^ { -1}
      \end{align*}
    \end{solution}

  \item
\begin{problem}[\vfill]
      More generally, if $h(x) = \dfrac{f(x)}{g(x)}$, we can rewrite
      $h(x)$ as $h(x) = f(x) [g(x)^{-1}]$.  Use the Product Rule and Chain
      Rule to differentiate this, and show that you get the same answer
      that the Quotient Rule gives you.
    \end{problem}

    \begin{solution}
      \begin{align*}
        \frac{d}{dx} \left(\fcolorbox{green}{white}{$\displaystyle f(x)$}\cdot \fcolorbox{green}{white}{$\displaystyle g(x)^ { -1} $}\right) = {} & f(x)\cdot \frac{d}{dx} \left(\fcolorbox{red}{white}{$\displaystyle \fcolorbox{blue}{white}{$\displaystyle g(x)$}^ { -1} $}\right)+ f'(x)\cdot g(x)^ { -1}  \\
        = {} & f(x)\cdot \left(-g(x)^ { -2} \cdot
          g'(x)\right)+ f'(x)\cdot g(x)^ { -1}
        \\
        = {} & - \frac{f(x) g'(x)}{g(x)^2} + \frac{f'(x)}{g(x)} \\
        = {} & \frac{g(x) f'(x) - f(x) g'(x)}{g(x)^2},
      \end{align*}
      which is exactly what the Quotient Rule says. 
    \end{solution}
  \end{enumerate}

\item \begin{problem}[\stretch{2}]
    An oil spill in a lagoon is forming an expanding circle.  At the current moment, the
    circle's radius is 30 meters, and the radius is increasing at an
    instantaneous rate of 5 meters per minute.  At what rate is the oil
    circle's area increasing (with respect to time)?  What are the units
    of your answer?

    \probonly{\emph{Make sure you understand why the Chain Rule is
        necessary here by first considering what quantities are related and how they are changing. Use the workbook problem about the rainwater tank to reason similarly about the area of the oil spill. What rate is given to you in the problem? What rate is asked for in this problem? How are these related using the Chain Rule?}} 
  \end{problem}

  \begin{solution}
    The area of a circle is given as a function of its radius by the
    formula $A = \pi r^2$.  However, the question is asking us for
    $\frac{dA}{dt}$, not $\frac{dA}{dr}$.  The problem tells us that the
    radius is increasing at an instantaneous rate of 5 m per minute. In
    mathematical notation, this means that $\frac{dr}{dt} = 5$ m/min at
    the instant when $r=30$.  In order to find $\frac{dA}{dt}$, we need
    the Chain Rule, which says $\frac{dA}{dt} =
    \frac{dA}{dr}\cdot\frac{dr}{dt}$. From the formula $A = \pi r^2$, we
    see that $\frac{dA}{dr} = 2\pi r$ m$^2$/m, or $60\pi$ m$^2$/m when
    $r=30$. Therefore $\frac{dA}{dt} = \frac{dA}{dr}\cdot\frac{dr}{dt} =
    (60\pi \text{ m$^2$/m} ) (5 \text{ m/min})= \boxed{300 \pi \text{
        m$^2$/min}}$. 
  \end{solution}
    
\end{enumerate}
